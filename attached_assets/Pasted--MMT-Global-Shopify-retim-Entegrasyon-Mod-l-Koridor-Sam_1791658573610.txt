import crypto from 'crypto';
import { db } from '../db/index.js'; 
import { approvalQueue } from '../db/schema.js'; 
import { eq, and } from 'drizzle-orm';

/**
 * 1. Adım: Ham Gövde (req.rawBody) ile Zaman Sabitli HMAC İmza Doğrulama
 */
export async function verifyShopifyWebhook(req, res, next) {
    try {
        const hmacHeader = req.get('X-Shopify-Hmac-Sha256');
        const secret = process.env.SHOPIFY_WEBHOOK_SECRET;

        if (!hmacHeader || !secret) {
            return res.status(401).json({ 
                status: 'error', 
                message: 'Çelik Kubbe: İmza veya gizli anahtar eksik.' 
            });
        }

        // Kesinlikle JSON.stringify kullanılmıyor; ham buffer/string veri baz alınıyor
        const rawBody = req.rawBody;
        if (!rawBody) {
            return res.status(400).json({ 
                status: 'error', 
                message: 'Çelik Kubbe: İstek ham gövdesi (raw body) doğrulanamadı.' 
            });
        }

        const calculatedHmac = crypto
            .createHmac('sha256', secret)
            .update(rawBody)
            .digest('base64');

        const bufferA = Buffer.from(calculatedHmac);
        const bufferB = Buffer.from(hmacHeader);

        if (bufferA.length !== bufferB.length || !crypto.timingSafeEqual(bufferA, bufferB)) {
            return res.status(403).json({ 
                status: 'error', 
                message: 'Çelik Kubbe: Geçersiz imza, erişim reddedildi.' 
            });
        }

        next();
    } catch (error) {
        return res.status(500).json({ status: 'error', message: 'İmza doğrulama hatası: ' + error.message });
    }
}

/**
 * 2. Adım: Drizzle ORM ve Sipariş/Güncelleme Duyarlı Güvenli Endpoint
 */
export async function handleShopifySync(req, res) {
    try {
        const { id: order_id, line_items, corridor, destination } = req.body;

        const targetCorridor = corridor || 'Samsun-Irak-Proje';
        if (!['Samsun-Irak-Proje', 'Samsun-Gurcistan'].includes(targetCorridor)) {
            return res.status(400).json({ 
                status: 'error', 
                message: 'Çelik Kubbe: Geçersiz lojistik koridoru.' 
            });
        }

        const orderIdStr = String(order_id || 'N/A');
        const itemTitle = line_items?.[0]?.title || 'Proje / Boru Sevkiyat Paketi';
        const quantity = line_items?.[0]?.quantity || 1;

        // Sipariş ve ürün bazlı mükerrer kontrolü (Aynı siparişin mükerrer işlenmesini önler)
        const existingRecord = await db.select()
            .from(approvalQueue)
            .where(
                and(
                    eq(approvalQueue.productId, orderIdStr),
                    eq(approvalQueue.approvalStatus, 'Kurucu Onayı Bekliyor')
                )
            )
            .limit(1);

        if (existingRecord.length > 0) {
            return res.status(200).json({
                status: 'success',
                message: 'Bu sipariş talebi halihazırda Kurucu Onay Kuyruğunda beklemektedir.',
                data: existingRecord[0]
            });
        }

        const transactionId = `MMT-SECURE-PROD-${Date.now()}`;

        // PostgreSQL veritabanına Drizzle ORM ile kalıcı kayıt
        const [insertedRecord] = await db.insert(approvalQueue).values({
            transactionId,
            productId: orderIdStr,
            itemTitle,
            stockCount: quantity,
            corridor: targetCorridor,
            destination: destination || 'Irak Şantiye Sahası',
            approvalStatus: 'Kurucu Onayı Bekliyor',
            authorizedBy: 'WelatTR (Onay Bekliyor)',
            createdAt: new Date()
        }).returning();

        console.log(`[MMT Nexus Üretim] Sipariş başarıyla işlendi ve kurucu onay kuyruğuna eklendi: ID ${orderIdStr}`);

        return res.status(200).json({
            status: 'success',
            message: 'Sistem güvenliğinde tam uyum sağlandı. Talep Kurucu Onay Kuyruğuna alındı.',
            data: insertedRecord
        });

    } catch (error) {
        console.error('Üretim Entegrasyon Hatası:', error.message);
        return res.status(500).json({ 
            status: 'error', 
            message: 'Veritabanı kayıt hatası: ' + error.message 
        });
    }
}
