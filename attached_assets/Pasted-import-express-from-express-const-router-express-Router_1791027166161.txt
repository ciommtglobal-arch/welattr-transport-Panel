import express from 'express';

const router = express.Router();

// Kalıcı veritabanı simülasyonu (Veritabanı bağlantınızla değiştirilebilir)
let FON_KREDI_HAVUZU = [];
let KURUMSAL_TALIMATLAR = [];

// 1. Çelik Kubbe & WelatTR İmzalı Fon ve Kredi Yönetim Modülü
router.post('/api/celik-kubbe/fon-islem', (req, res) => {
    const { islem_turu, miktar, kurum, aciklama } = req.body;

    const yeniIslem = {
        id: FON_KREDI_HAVUZU.length + 1,
        islem_turu: islem_turu || "Kredi / Fon Tahsisi",
        miktar: miktar || 0,
        kurum: kurum || "Borusan / Resmi Finans Kurumu",
        aciklama: aciklama,
        imza: "WelatTR",
        ekosistem: "Çelik Kubbe",
        guvence_durumu: "Resmi Mühürlü ve Onaylı",
        zaman: new Date().toISOString()
    };

    FON_KREDI_HAVUZU.push(yeniIslem);

    res.status(201).json({
        durum: "basarili",
        mesaj: "Fon ve kredi akışı Çelik Kubbe güvencesiyle mühürlendi.",
        veri: yeniIslem
    });
});

// 2. Resmi Kurum ve İş Ortağı Talimat Kayıt Modülü
router.post('/api/celik-kubbe/talimat-ekle', (req, res) => {
    const { kurum_adi, icerik, yetkili } = req.body;

    const yeniTalimat = {
        id: KURUMSAL_TALIMATLAR.length + 1,
        kurum: kurum_adi,
        icerik: icerik,
        yetkili: yetkili || "CİO / WelatTR",
        imza: "WelatTR",
        ekosistem: "Çelik Kubbe",
        zaman: new Date().toISOString()
    };

    KURUMSAL_TALIMATLAR.push(yeniTalimat);

    res.status(201).json({
        durum: "basarili",
        mesaj: "Kurumsal talimat resmi mühürle sisteme kaydedildi.",
        veri: yeniTalimat
    });
});

// 3. Havuz Listeleme Endpoint'i
router.get('/api/celik-kubbe/durum', (req, res) => {
    res.status(200).json({
        ekosistem: "Çelik Kubbe",
        temsilci: "WelatTR",
        toplam_fon_islem: FON_KREDI_HAVUZU.length,
        toplam_talimat: KURUMSAL_TALIMATLAR.length,
        fon_havuzu: FON_KREDI_HAVUZU,
        talimatlar: KURUMSAL_TALIMATLAR
    });
});

export default router;
