import json

# MMT Global - Pazar Yeri İkili Komisyon Paylaşım Modeli
MARKETPLACE_COMMISSION_RULES = {
    "Irak Hattı": {"toplam_komisyon_orani": 6.0, "max_risk": 5},
    "Dubai / BAE Hattı": {"toplam_komisyon_orani": 8.0, "max_risk": 7},
    "Avrupa Koridoru": {"toplam_komisyon_orani": 7.0, "max_risk": 4},
    "Ukrayna & Doğu Avrupa": {"toplam_komisyon_orani": 10.0, "max_risk": 8}
}

def pazar_yeri_komisyon_paylasimi_hesapla(koridor, musteri_satis_fiyati, risk_skoru):
    """
    Toplam komisyonun %50'sini yük verene (alıcıya), 
    %50'sini taşıyıcıya (satıcıya) paylaştıran otonom motor.
    """
    if koridor not in MARKETPLACE_COMMISSION_RULES:
        return {"durum": "HATA", "mesaj": "Geçersiz koridor tanımı."}
    
    kural = MARKETPLACE_COMMISSION_RULES[koridor]
    
    # Toplam platform komisyon tutarı (Satış fiyatı üzerinden)
    toplam_komisyon_tutari = musteri_satis_fiyati * (kural["toplam_komisyon_orani"] / 100.0)
    
    # %50 - %50 Paylaşım
    yuk_veren_payi = toplam_komisyon_tutari / 2.0
    tasiyici_payi = toplam_komisyon_tutari / 2.0
    
    risk_uygun = risk_skoru <= kural["max_risk"]
    
    if risk_uygun:
        return {
            "durum": "KOMİSYON PAYLAŞIMI UYGUN — SİSTEMSEL ÖN ONAY",
            "detaylar": {
                "koridor": koridor,
                "toplam_komisyon_usd": round(toplam_komisyon_tutari, 2),
                "yuk_veren_odemesi_50_yuzde": round(yuk_veren_payi, 2),
                "tasiyici_odemesi_50_yuzde": round(tasiyici_payi, 2),
                "hedef_butce_karsilama": "Personel ve Ofis Giderleri Havuzuna Aktarıldı"
            },
            "mesaj": f"Toplam ${round(toplam_komisyon_tutari, 2)} komisyon; ${round(yuk_veren_payi, 2)} Yük Veren, ${round(tasiyici_payi, 2)} Taşıyıcı olarak eşit bölüştürüldü."
        }
    else:
        return {
            "durum": "BLOKE / RİSK YÜKSEK",
            "mesaj": "Çelik Kubbe pazar yeri risk eşiği aşıldı."
        }

if __name__ == "__main__":
    # Örnek: Irak hattında 3000 USD satış fiyatı üzerinden %50-%50 komisyon paylaşımı testi
    sonuc = pazar_yeri_komisyon_paylasimi_hesapla("Irak Hattı", 3000.0, 3)
    print(json.dumps(sonuc, ensure_ascii=False, indent=2))
