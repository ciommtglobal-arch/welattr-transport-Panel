import os
import json
from decimal import Decimal, ROUND_HALF_UP

# --- MMT GLOBAL ANA SİSTEM ENTEGRASYON YÖNETİCİSİ ---
SISTEM_AYARLARI = {
    "proje": "MMT Connect & Pazar Yeri",
    "kurucu": "Mehmet Alioğlu (CIO / WelatTR)",
    "cerceve": "Çelik Kubbe Risk & Marj Yönetimi"
}

MARKETPLACE_TARIFF = {
    "Irak": {"toplam_komisyon_orani": Decimal("6.0"), "max_risk": 5},
    "Dubai / BAE": {"toplam_komisyon_orani": Decimal("8.0"), "max_risk": 7},
    "Avrupa": {"toplam_komisyon_orani": Decimal("7.0"), "max_risk": 4},
    "Ukrayna / Doğu Avrupa": {"toplam_komisyon_orani": Decimal("10.0"), "max_risk": 8}
}

def mmt_sistem_islem_isleyici(koridor, musteri_satis_fiyati, risk_skoru):
    """
    Tüm sistemi birbirine bağlayan ana otonom işleyici:
    1. Geçersiz/sıfır/negatif değerleri engeller.
    2. Çelik Kubbe risk eşiğini denetler.
    3. Decimal ile kuruş hassasiyetinde %50 - %50 komisyon dağılımı yapar.
    4. Telegram ve arayüz entegrasyonu için taslak iç kontrol raporu üretir.
    """
    try:
        satis = Decimal(str(musteri_satis_fiyati))
        risk = int(risk_skoru)
    except (ValueError, TypeError):
        return {"durum": "HATA", "mesaj": "Geçersiz veri biçimi."}
    
    if satis <= Decimal("0.00") or risk < 0:
        return {"durum": "REDDEDİLDİ", "mesaj": "Sıfır veya negatif değer kabul edilmez."}
    
    if koridor not in MARKETPLACE_TARIFF:
        return {"durum": "HATA", "mesaj": "Tanımsız lojistik koridoru."}
    
    kural = MARKETPLACE_TARIFF[koridor]
    
    # Çelik Kubbe Risk Denetimi
    if risk > kural["max_risk"]:
        return {
            "durum": "BLOKE / ÇELİK KUBBE RİSK UYARISI",
            "mesaj": f"Risk skoru ({risk}) izin verilen maksimum sınırın ({kural['max_risk']}) üzerinde."
        }
    
    # Kuruş Hassasiyetinde Komisyon Hesabı
    toplam_komisyon = (satis * (kural["toplam_komisyon_orani"] / Decimal("100.0"))).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
    yuk_veren_payi = (toplam_komisyon / Decimal("2.0")).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
    tasiyici_payi = toplam_komisyon - yuk_veren_payi
    
    return {
        "durum": "SİSTEMSEL ÖN KONTROL UYGUN — KURUCU ONAYI BEKLENİYOR",
        "sistem_bilgisi": SISTEM_AYARLARI,
        "detaylar": {
            "koridor": koridor,
            "musteri_satis_fiyati_usd": float(satis),
            "toplam_komisyon_usd": float(toplam_komisyon),
            "yuk_veren_odemesi_50": float(yuk_veren_payi),
            "tasiyici_odemesi_50": float(tasiyici_payi),
            "kuruş_dogrulama": float(yuk_veren_payi + tasiyici_payi) == float(toplam_komisyon)
        }
    }

if __name__ == "__main__":
    # Sistem Test Simülasyonu

    print(json.dumps(test_sonucu, ensure_ascii=False, indent=2))
