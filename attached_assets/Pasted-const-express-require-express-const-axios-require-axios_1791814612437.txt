const express = require('express');
const axios = require('axios');
const app = express();

app.use(express.json());

// Çevre değişkenlerinden (Replit Secrets) gerekli anahtarları alıyoruz
const VERIFY_TOKEN = process.env.META_VERIFY_TOKEN;
const WHATSAPP_TOKEN = process.env.WHATSAPP_ACCESS_TOKEN;
const PHONE_NUMBER_ID = process.env.WHATSAPP_PHONE_NUMBER_ID;

// 1. Meta Webhook Doğrulama (GET)
app.get('/webhook', (req, res) => {
  const mode = req.query['hub.mode'];
  const token = req.query['hub.verify_token'];
  const challenge = req.query['hub.challenge'];

  if (mode && token) {
    if (mode === 'subscribe' && token === VERIFY_TOKEN) {
      console.log('WEBHOOK_VERIFIED: Meta webhook başarıyla doğrulandı.');
      res.status(200).send(challenge);
    } else {
      res.sendStatus(403);
    }
  } else {
    res.sendStatus(400);
  }
});

// 2. Meta / WhatsApp Üzerinden Gelen Mesajlar (POST)
app.post('/webhook', async (req, res) => {
  try {
    const body = req.body;

    if (body.object === 'whatsapp_business_account') {
      body.entry.forEach(entry => {
        entry.changes.forEach(change => {
          if (change.value.messages && change.value.messages[0]) {
            const senderID = change.value.messages[0].from;
            const messageBody = change.value.messages[0].text.body;

            console.log(`Yeni WhatsApp mesajı - Gönderen: ${senderID}, Mesaj: ${messageBody}`);
            processLogisticsOrder(senderID, messageBody);
          }
        });
      });
      res.status(200).send('EVENT_RECEIVED');
    } else {
      res.sendStatus(404);
    }
  } catch (error) {
    console.error('Webhook işleme hatası:', error);
    res.sendStatus(500);
  }
});

// 3. Shopify Yeni Sipariş Webhook Entegrasyonu (POST)
app.post('/shopify-webhook/orders/create', async (req, res) => {
  try {
    const order = req.body;
    
    const orderId = order.id;
    const customerName = order.customer ? `${order.customer.first_name} ${order.customer.last_name}` : 'Misafir Müşteri';
    const shippingAddress = order.shipping_address ? `${order.shipping_address.city}, ${order.shipping_address.country}` : 'Adres belirtilmemiş';
    const lineItems = order.line_items ? order.line_items.map(item => `${item.quantity}x ${item.name}`).join(', ') : 'Ürün bilgisi yok';

    console.log(`[SHOPIFY] Yeni Sipariş Alındı! #${orderId} - Müşteri: ${customerName} - Ürünler: ${lineItems}`);

    // Siparişi lojistik ağına ve veritabanına işleme
    await processShopifyLogistics(orderId, customerName, shippingAddress, lineItems);

    res.status(200).send('Shopify siparişi başarıyla alındı ve lojistiğe aktarıldı.');
  } catch (error) {
    console.error('Shopify webhook işleme hatası:', error);
    res.sendStatus(500);
  }
});

// Lojistik Ağına Aktarım Fonksiyonları
async function processLogisticsOrder(customerPhone, orderDetails) {
  console.log(`Lojistik ağı güncelleniyor -> Müşteri: ${customerPhone}, Talep: ${orderDetails}`);
}

async function processShopifyLogistics(orderId, customerName, address, items) {
  console.log(`Shopify Lojistik Rotası -> Sipariş #${orderId} (${customerName}) rotaya işlendi. Adres: ${address}`);
}

// Sunucuyu Başlatma
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`MMT Global Entegrasyon sunucusu ${PORT} portunda aktif.`);
});
