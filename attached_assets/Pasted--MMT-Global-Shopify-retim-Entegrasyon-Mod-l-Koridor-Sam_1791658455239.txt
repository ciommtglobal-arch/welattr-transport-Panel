import crypto from 'crypto';
import { db } from '../db/index.js'; // Projenizin Drizzle DB bağlantı yolu
import { approvalQueue } from '../db/schema.js'; // PostgreSQL / Drizzle tablo şemanız
import { eq } from 'drizzle-orm';

/**
 * 1. Adım: Ham Gövde (Raw Body) ile Zaman Sabitli HMAC İmza Doğrulama
 */
export async function verifyShopifyWebhook(req, res, next) {
    try {
        const hmacHeader = req.get('X-Shopify-Hmac-Sha256');
        const secret = process.env.SHOPIFY_WEBHOOK_SECRET;

        if (!hmacHeader || !secret) {
            return res.status(401).json({ 
                status: 'error', 
                message: 'Çelik Kubbe: İmza veya gizli anahtar eksik.' 
            });
        }

        // Shopify webhook için req.rawBody (buffer veya string) kullanımı şarttır
        const rawBody = req.rawBody || JSON.stringify(req.body);
        const calculatedHmac = crypto
            .createHmac('sha256', secret)
            .update(rawBody)
            .digest('base64');

        const bufferA = Buffer.from(calculatedHmac);
        const bufferB = Buffer.from(hmacHeader);

        if (bufferA.length !== bufferB.length || !crypto.timingSafeEqual(bufferA, bufferB)) {
            return res.status(403).json({ 
                status: 'error', 
                message: 'Çelik Kubbe: Geçersiz imza, erişim reddedildi.' 
            });
        }

        next();
    } catch (error) {
        return res.status(500).json({ status: 'error', message: 'İmza doğrulama hatası: ' + error.message });
    }
}

/**
 * 2. Adım: Drizzle ORM ve PostgreSQL Uyumlu Güvenli Endpoint
 */
export async function handleShopifySync(req, res) {
    try {
        const { product_id, title, inventory_quantity, corridor, destination } = req.body;

        const targetCorridor = corridor || 'Samsun-Irak-Proje';
        if (!['Samsun-Irak-Proje', 'Samsun-Gurcistan'].includes(targetCorridor)) {
            return res.status(400).json({ 
                status: 'error', 
                message: 'Çelik Kubbe: Geçersiz lojistik koridoru.' 
            });
        }

        const productIdStr = String(product_id || 'N/A');

        // Mükerrer kayıt (Duplicate) kontrolü
        const existingRecord = await db.select()
            .from(approvalQueue)
            .where(eq(approvalQueue.productId, productIdStr))
            .limit(1);

        if (existingRecord.length > 0 && existingRecord[0].approvalStatus === 'Kurucu Onayı Bekliyor') {
            return res.status(200).json({
                status: 'success',
                message: 'Bu talep zaten Kurucu Onay Kuyruğunda bekliyor.',
                data: existingRecord[0]
            });
        }

        const transactionId = `MMT-SECURE-PG-${Date.now()}`;

        // Drizzle ORM ile PostgreSQL veritabanına kalıcı kayıt
        const [insertedRecord] = await db.insert(approvalQueue).values({
            transactionId,
            productId: productIdStr,
            itemTitle: title || 'Proje / Boru Sevkiyat Kalemi',
            stockCount: inventory_quantity || 0,
            corridor: targetCorridor,
            destination: destination || 'Irak Şantiye Sahası',
            approvalStatus: 'Kurucu Onayı Bekliyor',
            authorizedBy: 'WelatTR (Beklemede)',
            createdAt: new Date()
        }).returning();

        console.log(`[MMT Nexus PostgreSQL] Başarıyla işlendi ve onay kuyruğuna eklendi: ${title}`);

        return res.status(200).json({
            status: 'success',
            message: 'Veritabanı kalıcılığı (Drizzle/PostgreSQL) sağlandı. Talep Kurucu Onay Kuyruğuna eklendi.',
            data: insertedRecord
        });

    } catch (error) {
        console.error('Veritabanı Entegrasyon Hatası:', error.message);
        return res.status(500).json({ 
            status: 'error', 
            message: 'Veritabanı kayıt hatası: ' + error.message 
        });
    }
}
