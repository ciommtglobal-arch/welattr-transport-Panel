import os
import json
from decimal import Decimal, ROUND_HALF_UP

# --- MMT GLOBAL - TAM OTONOM PAZAR YERİ VE ÇELİK KUBBE MOTORU ---
SISTEM_KAYIT = {
    "proje": "MMT Connect & Çelik Kubbe",
    "kurucu": "Mehmet Alioğlu (CIO / WelatTR)",
    "mod": "Tam Otonom / Personelsiz İzleme"
}

MARKETPLACE_TARIFF = {
    "Irak": {"toplam_komisyon_orani": Decimal("6.0"), "max_risk": 5},
    "Dubai / BAE": {"toplam_komisyon_orani": Decimal("8.0"), "max_risk": 7},
    "Avrupa": {"toplam_komisyon_orani": Decimal("7.0"), "max_risk": 4},
    "Ukrayna / Doğu Avrupa": {"toplam_komisyon_orani": Decimal("10.0"), "max_risk": 8}
}

def mmt_otonom_islem_calistir(koridor, musteri_satis_fiyati, tirsahibi_maliyeti, risk_skoru):
    """
    1. Çelik Kubbe ile risk ve sınır denetimi yapar.
    2. Decimal ile kuruş hatasız komisyon ve marj hesaplar.
    3. Tır sahibinin hakedişinden komisyonu düşerek net ödemeyi belirler.
    4. Telegram cüzdan hattı için bildirim verisi hazırlar.
    """
    try:
        satis = Decimal(str(musteri_satis_fiyati))
        maliyet = Decimal(str(tirsahibi_maliyeti))
        risk = int(risk_skoru)
    except (ValueError, TypeError):
        return {"durum": "HATA", "mesaj": "Geçersiz veri biçimi."}
    
    if satis <= Decimal("0.00") or maliyet <= Decimal("0.00") or risk < 0:
        return {"durum": "REDDEDİLDİ", "mesaj": "Geçersiz fiyat veya risk değeri."}
    
    if koridor not in MARKETPLACE_TARIFF:
        return {"durum": "HATA", "mesaj": "Tanımsız lojistik koridoru."}
    
    kural = MARKETPLACE_TARIFF[koridor]
    
    # Çelik Kubbe Risk Süzgeci
    if risk > kural["max_risk"]:
        return {
            "durum": "BLOKE / ÇELİK KUBBE RİSK AŞIMI",
            "mesaj": f"Risk skoru ({risk}) izin verilen maksimum sınırın ({kural['max_risk']}) üzerinde."
        }
    
    # Ticari Marj ve Komisyon Hesapları
    toplam_komisyon = (satis * (kural["toplam_komisyon_orani"] / Decimal("100.0"))).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
    yuk_veren_komisyon_payi = (toplam_komisyon / Decimal("2.0")).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
    tasiyici_komisyon_payi = toplam_komisyon - yuk_veren_komisyon_payi
    
    # Tır Sahibi Hakediş Kesintisi
    tir_sahibi_brut = maliyet
    tir_sahibi_net_hakedis = tir_sahibi_brut - tasiyici_komisyon_payi
    
    # Alıcının Toplam Ödemesi
    alicinin_toplam_odemesi = maliyet + yuk_veren_komisyon_payi
    
    return {
        "durum": "OTONOM AKIŞ ONAYLANDI - TELEGRAM CÜZDANA AKTARILDI",
        "sistem_bilgisi": SISTEM_KAYIT,
        "detaylar": {
            "koridor": koridor,
            "alicinin_toplam_odemesi_usd": float(alicinin_toplam_odemesi),
            "tir_sahibinin_net_hakedisi_usd": float(tir_sahibi_net_hakedis),
            "telegram_cuzdana_geçecek_komisyon_usd": float(toplam_komisyon),
            "ticari_satis_farki_ayirici": float(satis - maliyet)
        }
    }

if __name__ == "__main__":
    # Canlı Simülasyon Testi
    islem_sonucu = mmt_otonom_islem_calistir("Irak", 3000.0, 2700.0, 3)
    print(json.dumps(islem_sonucu, ensure_ascii=False, indent=2))
