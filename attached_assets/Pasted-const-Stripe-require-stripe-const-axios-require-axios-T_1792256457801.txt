const Stripe = require('stripe');
const axios = require('axios'); // Tami REST API haberleşmesi için
const { db } = require('./db');
const { orders } = require('./schema');
const { eq, and } = require('drizzle-orm');

// 1. Tami Resmi Yapılandırması (Environment Secrets üzerinden)
const tamiMerchantNumber = process.env.TAMI_MERCHANT_NUMBER || '';
const tamiSecretKey = process.env.TAMI_SECRET_KEY || '';
const isTamiSandbox = !process.env.NODE_ENV || process.env.NODE_ENV !== 'production';
const tamiBaseUrl = isTamiSandbox ? 'https://sandbox-api.tami.com.tr' : 'https://api.tami.com.tr';

// 2. Stripe Küresel Yapılandırması (Uluslararası / Döviz)
const stripe = new Stripe(process.env.STRIPE_SECRET_KEY, {
    apiVersion: '2023-10-16',
});

function registerUnifiedPaymentRoutes(app) {
    
    // ORTAK ÖDEME BAŞLATMA ROUTE'U (Gateway: 'tami' veya 'stripe')
    app.post('/api/payment/initialize-unified', async (req, res) => {
        try {
            // Clerk Oturum Güvenlik Kontrolü
            const authHeader = req.headers.authorization;
            if (!authHeader || !authHeader.startsWith('Bearer ')) {
                return res.status(401).json({
                    status: 'error',
                    message: 'Yetkilendirme hatası: Geçerli Clerk oturum tokeni gereklidir.'
                });
            }

            const { orderId, gateway } = req.body;
            if (!orderId || !gateway) {
                return res.status(400).json({
                    status: 'error',
                    message: 'Eksik parametre: orderId ve gateway (tami/stripe) zorunludur.'
                });
            }

            // Veritabanından siparişi güvenle çek
            const existingOrders = await db.select().from(orders).where(eq(orders.id, orderId)).limit(1);
            if (!existingOrders || existingOrders.length === 0) {
                return res.status(404).json({
                    status: 'error',
                    message: 'Belirtilen sipariş veritabanında bulunamadı.'
                });
            }

            const dbOrder = existingOrders[0];

            // Mükerrer Ödeme Koruması
            if (dbOrder.status === 'PAID') {
                return res.status(400).json({
                    status: 'error',
                    message: 'Bu sipariş daha önceden başarıyla ödenmiştir.'
                });
            }

            // SEÇENEK A: YURT İÇİ TAMI POS AKIŞI
            if (gateway === 'tami') {
                // Tami API istek payload yapısı
                const tamiPayload = {
                    merchantNumber: tamiMerchantNumber,
                    orderId: dbOrder.id.toString(),
                    amount: parseFloat(dbOrder.totalAmount).toFixed(2),
                    currency: 'TRY',
                    callbackUrl: `${process.env.APP_BASE_URL || 'https://panel.mmtglobal.com.tr'}/api/payment/tami/callback`,
                    buyer: {
                        name: dbOrder.buyerName || 'Mehmet Alioğlu',
                        email: dbOrder.buyerEmail || 'cio@mmtglobal.com.tr',
                        gsmNumber: dbOrder.buyerGsm || '+905000000000'
                    }
                };

                // Tami sunucusuna güvenli istek
                try {
                    const tamiResponse = await axios.post(`${tamiBaseUrl}/v1/payment/init`, tamiPayload, {
                        headers: {
                            'Authorization': `Bearer ${tamiSecretKey}`,
                            'Content-Type': 'application/json'
                        }
                    });

                    const paymentData = tamiResponse.data;

                    // Token / İşlem ID kaydı
                    await db.update(orders)
                        .set({ paymentToken: paymentData.token || paymentData.paymentId, paymentGateway: 'tami', updatedAt: new Date() })
                        .where(eq(orders.id, orderId));

                    return res.json({
                        status: 'SUCCESS',
                        gateway: 'tami',
                        token: paymentData.token,
                        paymentPageUrl: paymentData.paymentPageUrl
                    });

                } catch (tamiErr) {
                    return res.status(400).json({
                        status: 'FAILURE',
                        message: 'Tami API işlem hatası: ' + (tamiErr.response?.data?.message || tamiErr.message)
                    });
                }
            } 
            // SEÇENEK B: ULUSLARARASI STRIPE AKIŞI
            else if (gateway === 'stripe') {
                const amountInCents = Math.round(parseFloat(dbOrder.totalAmount) * 100);

                const session = await stripe.checkout.sessions.create({
                    payment_method_types: ['card'],
                    line_items: [
                        {
                            price_data: {
                                currency: (dbOrder.currency || 'usd').toLowerCase(),
                                product_data: {
                                    name: dbOrder.itemName || 'MMT Global International Logistics Service',
                                    description: `Order ID: ${dbOrder.id}`,
                                },
                                unit_amount: amountInCents,
                            },
                            quantity: 1,
                        },
                    ],
                    mode: 'payment',
                    success_url: `${process.env.APP_BASE_URL || 'https://panel.mmtglobal.com.tr'}/odeme-basarili?session_id={CHECKOUT_SESSION_ID}&orderId=${dbOrder.id}`,
                    cancel_url: `${process.env.APP_BASE_URL || 'https://panel.mmtglobal.com.tr'}/odeme-basarisiz?orderId=${dbOrder.id}`,
                    client_reference_id: dbOrder.id.toString(),
                    metadata: { orderId: dbOrder.id.toString(), gateway: 'stripe' }
                });

                await db.update(orders)
                    .set({ paymentToken: session.id, paymentGateway: 'stripe', updatedAt: new Date() })
                    .where(eq(orders.id, orderId));

                return res.json({
                    status: 'SUCCESS',
                    gateway: 'stripe',
                    sessionId: session.id,
                    url: session.url
                });
            } else {
                return res.status(400).json({ status: 'error', message: 'Geçersiz ödeme sağlayıcısı seçildi.' });
            }

        } catch (error) {
            res.status(500).json({
                status: 'error',
                message: 'Sunucu iç hatası: ' + error.message
            });
        }
    });

    // TAMI CALLBACK ENDPOINTİ
    app.post('/api/payment/tami/callback', async (req, res) => {
        try {
            const { orderId, status, amount } = req.body;
            if (!orderId || status !== 'SUCCESS') {
                return res.redirect('/odeme-basarisiz?reason=declined');
            }

            // Mükerrer işlem ve tutar korumalı veritabanı güncellemesi
            await db.update(orders)
                .set({ status: 'PAID', paidPrice: amount, updatedAt: new Date() })
                .where(and(eq(orders.id, orderId), eq(orders.status, 'PENDING')));

            return res.redirect(`/odeme-basarili?orderId=${orderId}&price=${amount}`);
        } catch (error) {
            return res.redirect('/odeme-basarisiz?error=' + encodeURIComponent(error.message));
        }
    });
}

module.exports = registerUnifiedPaymentRoutes;
