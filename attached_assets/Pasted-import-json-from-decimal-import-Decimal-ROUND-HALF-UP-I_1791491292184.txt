import json
from decimal import Decimal, ROUND_HALF_UP, InvalidOperation

SISTEM_KAYIT = {
    "proje": "MMT Connect & Çelik Kubbe",
    "kurucu": "Mehmet Alioğlu (CIO / WelatTR)",
    "durum": "TASLAK – CANLIYA ALINMADI"
}

MARKETPLACE_TARIFF = {
    "Irak": {"toplam_komisyon_orani": Decimal("6.0"), "max_risk": 5},
    "Dubai / BAE": {"toplam_komisyon_orani": Decimal("8.0"), "max_risk": 7},
    "Avrupa": {"toplam_komisyon_orani": Decimal("7.0"), "max_risk": 4},
    "Ukrayna / Doğu Avrupa": {"toplam_komisyon_orani": Decimal("10.0"), "max_risk": 8}
}

def mmt_kusursuz_hesaplayici(koridor, musteri_satis_fiyati, tirsahibi_maliyeti, risk_skoru):
    try:
        satis = Decimal(str(musteri_satis_fiyati)).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
        maliyet = Decimal(str(tirsahibi_maliyeti)).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
        
        if not (satis.is_finite() and maliyet.is_finite()):
            return {"durum": "HATA", "mesaj": "Sayısal değerler sonlu olmalıdır."}
            
        risk_dec = Decimal(str(risk_skoru))
        if not risk_dec.is_finite() or risk_dec % 1 != 0:
            return {"durum": "REDDEDİLDİ", "mesaj": "Risk skoru sonlu bir tam sayı olmalıdır."}
        risk = int(risk_dec)
        
        if satis <= Decimal("0.00") or maliyet <= Decimal("0.00") or risk < 0:
            return {"durum": "REDDEDİLDİ", "mesaj": "Geçersiz fiyat veya negatif risk değeri."}
        
        if koridor not in MARKETPLACE_TARIFF:
            return {"durum": "HATA", "mesaj": "Tanımsız lojistik koridoru."}
        
        kural = MARKETPLACE_TARIFF[koridor]
        if risk > kural["max_risk"]:
            return {"durum": "BLOKE", "mesaj": f"Çelik Kubbe risk eşiği aşıldı (Risk: {risk})"}
        
        # Hesaplama ve Dağıtım Blokları
        toplam_komisyon = (satis * (kural["toplam_komisyon_orani"] / Decimal("100.0"))).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
        tasiyici_komisyon_payi = (toplam_komisyon / Decimal("2.0")).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
        yuk_veren_komisyon_payi = toplam_komisyon - tasiyici_komisyon_payi
        
        brut_ticari_satis_farki = (satis - maliyet).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
        if brut_ticari_satis_farki < Decimal("0.00"):
            return {"durum": "BLOKE", "mesaj": "Satış fiyatı maliyetten düşük olamaz."}
            
        tir_sahibi_net_hakedis = (maliyet - tasiyici_komisyon_payi).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
        if tir_sahibi_net_hakedis < Decimal("0.00"):
            return {"durum": "BLOKE", "mesaj": "Taşıyıcı net hakedişi negatif olamaz."}
            
        alicinin_toplam_odemesi = (maliyet + yuk_veren_komisyon_payi).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)

    except (ValueError, TypeError, InvalidOperation, ArithmeticError):
        return {"durum": "HATA", "mesaj": "Matematiksel işlem veya veri biçimi hatası."}

    return {
        "durum": "TASLAK HESAP - KURUCU ONAYI BEKLENİYOR",
        "sistem_bilgisi": SISTEM_KAYIT,
        "detaylar": {
            "koridor": koridor,
            "alicinin_toplam_odemesi": str(alicinin_toplam_odemesi),
            "tir_sahibi_net_hakedis": str(tir_sahibi_net_hakedis),
            "mmt_platform_komisyonu": str(toplam_komisyon),
            "mmt_brut_ticari_satis_farki": str(brut_ticari_satis_farki)
        }
    }

if __name__ == "__main__":
    sonuc = mmt_kusursuz_hesaplayici("Irak", "3000.00", "2700.00", 3)
    print(json.dumps(sonuc, ensure_ascii=False, indent=2))
