const express = require('express');
const crypto = require('crypto');
const app = express();

app.use(express.json());

// Çevre değişkenlerinden (Replit Secrets) gerekli anahtarları alıyoruz
const VERIFY_TOKEN = process.env.META_VERIFY_TOKEN;
const SHOPIFY_SECRET = process.env.SHOPIFY_WEBHOOK_SECRET;

// 1. Meta Webhook Doğrulama (GET)
app.get('/webhook', (req, res) => {
  const mode = req.query['hub.mode'];
  const token = req.query['hub.verify_token'];
  const challenge = req.query['hub.challenge'];

  if (mode && token) {
    if (mode === 'subscribe' && token === VERIFY_TOKEN) {
      console.log('WEBHOOK_VERIFIED: Meta webhook başarıyla doğrulandı.');
      res.status(200).send(challenge);
    } else {
      res.sendStatus(403);
    }
  } else {
    res.sendStatus(400);
  }
});

// 2. Meta / WhatsApp Üzerinden Gelen Mesajlar (POST)
app.post('/webhook', async (req, res) => {
  try {
    const body = req.body;

    if (body.object === 'whatsapp_business_account') {
      body.entry.forEach(entry => {
        entry.changes.forEach(change => {
          if (change.value.messages && change.value.messages[0]) {
            const senderID = change.value.messages[0].from;
            const messageBody = change.value.messages[0].text ? change.value.messages[0].text.body : '[Metin dışı mesaj]';

            console.log(`[WelatTR] Yeni WhatsApp talebi - Gönderen: ${senderID}, İçerik: ${messageBody}`);
            processLogisticsRequest(senderID, messageBody);
          }
        });
      });
      res.status(200).send('EVENT_RECEIVED');
    } else {
      res.sendStatus(404);
    }
  } catch (error) {
    console.error('Meta webhook işleme hatası:', error);
    res.sendStatus(500);
  }
});

// 3. Shopify Yeni Sipariş Webhook Entegrasyonu (POST) ve Güvenlik İmza Kontrolü
app.post('/shopify-webhook/orders/create', async (req, res) => {
  try {
    const hmacHeader = req.get('X-Shopify-Hmac-Sha256');
    // Güvenlik: İmza doğrulaması (Çelik Kubbe disiplini)
    if (SHOPIFY_SECRET && hmacHeader) {
      const generatedHmac = crypto
        .createHmac('sha256', SHOPIFY_SECRET)
        .update(JSON.stringify(req.body), 'utf8')
        .digest('base64');
      
      if (generatedHmac !== hmacHeader) {
        console.warn('[SECURITY] Geçersiz Shopify imza denemesi engellendi.');
        return res.status(401).send('Geçersiz imza.');
      }
    }

    const order = req.body;
    const orderId = order.id;
    const customerName = order.customer ? `${order.customer.first_name} ${order.customer.last_name}` : 'Müşteri';
    const shippingAddress = order.shipping_address ? `${order.shipping_address.city}, ${order.shipping_address.country}` : 'Adres yok';
    const lineItems = order.line_items ? order.line_items.map(item => `${item.quantity}x ${item.name}`).join(', ') : 'Ürün yok';

    console.log(`[SHOPIFY] Yeni Sipariş Alındı #${orderId} - Müşteri: ${customerName} - Lokasyon: ${shippingAddress}`);

    // Siparişi lojistik ağına işleme fonksiyonu
    await processShopifyLogistics(orderId, customerName, shippingAddress, lineItems);

    res.status(200).send('Shopify siparişi güvenle alındı ve lojistiğe işlendi.');
  } catch (error) {
    console.error('Shopify webhook işleme hatası:', error);
    res.sendStatus(500);
  }
});
