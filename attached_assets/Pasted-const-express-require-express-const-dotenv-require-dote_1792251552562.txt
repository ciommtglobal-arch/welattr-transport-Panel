const express = require('express');
const dotenv = require('dotenv');
const Iyzipay = require('iyzipay');
// Not: Projenizde kullandığınız veritabanı/Drizzle ORM bağlantınızı buraya dahil edin:
// const { db } = require('./db'); // Örnek Drizzle veya PostgreSQL bağlantısı

dotenv.config();

const app = express();
const PORT = process.env.PORT || 3000;

app.use(express.json());

// İyzico İstemcisi Yapılandırması (.env anahtarlarıyla ve ortam seçimiyle)
const iyzipay = new Iyzipay({
    apiKey: process.env.IYZICO_API_KEY,
    secretKey: process.env.IYZICO_SECRET_KEY,
    uri: process.env.IYZICO_BASE_URI || 'https://sandbox-api.iyzipay.com' // Canlı için https://api.iyzipay.com
});

// Sağlık Kontrolü
app.get('/', (req, res) => {
    res.json({
        status: 'success',
        message: 'MMT Connect - Güvenli ve Doğrulanmış İyzico Ödeme Servisi Aktif'
    });
});

// 1. Güvenli Ödeme Başlatma Endpoint'i (Gerçek Veritabanı ve Clerk Doğrulamalı)
app.post('/api/payment/initialize', async (req, res) => {
    try {
        // Clerk Kimlik Doğrulama Kontrolü (Bearer Token)
        const authHeader = req.headers.authorization;
        if (!authHeader || !authHeader.startsWith('Bearer ')) {
            return res.status(401).json({
                status: 'error',
                message: 'Yetkilendirme hatası: Geçerli bir Clerk oturum tokeni bulunamadı.'
            });
        }
        
        const clerkToken = authHeader.split(' ')[1];
        // Not: Gerçek projede burada Clerk SDK ile token doğrulanır ve userId alınır.
        // Örn: const session = await clerk.verifyToken(clerkToken); const userId = session.sub;

        const { orderId } = req.body;
        if (!orderId) {
            return res.status(400).json({
                status: 'error',
                message: 'Sipariş ID (orderId) belirtilmesi zorunludur.'
            });
        }

        // GÜVENLİK KURALI: Tutar, sepet ve adresler dışarıdan (client tarafı) asla alınmaz!
        // Doğrudan veritabanından orderId'ye karşılık gelen gerçek sipariş ve kullanıcı bilgileri çekilir.
        /* 
           Örnek Veritabanı Sorgusu:
           const order = await db.select().from(orders).where(eq(orders.id, orderId)).limit(1);
           if (!order || order.length === 0) return res.status(404).json({ message: 'Sipariş bulunamadı' });
           const dbOrder = order[0];
        */

        // Güvenlik simülasyonu (Gerçek uygulamada yukarıdaki veritabanı sorgusu aktif edilir):
        const mockDbOrder = {
            id: orderId,
            totalAmount: '1250.00', // Veritabanındaki kesin tutar
            currency: 'TRY',
            buyer: {
                id: 'USER-CLERK-987',
                name: 'Mehmet',
                surname: 'Alioğlu',
                email: 'cio@mmtglobal.com.tr',
                gsmNumber: '+905000000000',
                identityNumber: '11111111111',
                registrationAddress: 'MMT Global Lojistik Merkez Cizre/Şırnak',
                city: 'Şırnak',
                country: 'Turkey',
                zipCode: '73200'
            },
            items: [
                { id: 'PROD-101', name: 'Ağır Vasıta Yedek Parça / Dorse Ekipmanı', category1: 'Lojistik Donanım', price: '1250.00' }
            ]
        };

        const request = {
            locale: Iyzipay.LOCALE.TR,
            conversationId: mockDbOrder.id,
            price: mockDbOrder.totalAmount,
            paidPrice: mockDbOrder.totalAmount,
            currency: Iyzipay.CURRENCY.TRY,
            basketId: mockDbOrder.id,
            paymentGroup: Iyzipay.PAYMENT_GROUP.PRODUCT,
            callbackUrl: `${process.env.APP_BASE_URL || 'https://panel.mmtglobal.com.tr'}/api/payment/callback`,
            buyer: {
                id: mockDbOrder.buyer.id,
                name: mockDbOrder.buyer.name,
                surname: mockDbOrder.buyer.surname,
                gsmNumber: mockDbOrder.buyer.gsmNumber,
                email: mockDbOrder.buyer.email,
                identityNumber: mockDbOrder.buyer.identityNumber,
                lastLoginDate: '2026-10-17 18:38:00',
                registrationDate: '2026-01-01 12:00:00',
                registrationAddress: mockDbOrder.buyer.registrationAddress,
                ip: req.ip || '85.105.0.1',
                city: mockDbOrder.buyer.city,
                country: mockDbOrder.buyer.country,
                zipCode: mockDbOrder.buyer.zipCode
            },
            shippingAddress: {
                contactName: `${mockDbOrder.buyer.name} ${mockDbOrder.buyer.surname}`,
                city: mockDbOrder.buyer.city,
                country: mockDbOrder.buyer.country,
                address: mockDbOrder.buyer.registrationAddress,
                zipCode: mockDbOrder.buyer.zipCode
            },
            billingAddress: {
                contactName: `${mockDbOrder.buyer.name} ${mockDbOrder.buyer.surname}`,
                city: mockDbOrder.buyer.city,
                country: mockDbOrder.buyer.country,
                address: mockDbOrder.buyer.registrationAddress,
                zipCode: mockDbOrder.buyer.zipCode
            },
            basketItems: mockDbOrder.items.map(item => ({
                id: item.id,
                name: item.name,
                category1: item.category1,
                itemType: Iyzipay.BASKET_ITEM_TYPE.PHYSICAL,
                price: item.price
            }))
        };

        // İyzico Ödeme Formu Oluşturma Servisi
        iyzipay.checkoutFormInitialize.create(request, (err, result) => {
            if (err) {
                return res.status(500).json({
                    status: 'error',
                    message: 'İyzico ağ bağlantı hatası',
                    error: err.message
                });

                    status: 'failure',

                    errorMessage: result?.errorMessage || 'İyzico ödeme formu başlatılamadı.'
                });
            }

            // Başarılı form içeriği ve token dönülür
            res.json({
                status: 'success',
                token: result.token,
                checkoutFormContent: result.checkoutFormContent,
                paymentPageUrl: result.paymentPageUrl,
                conversationId: mockDbOrder.id
            });
        });

    } catch (error) {
        res.status(500).json({
            status: 'error',
            message: 'Sunucu iç hatası: ' + error.message
        });
    }
});

// 2. Ödeme Sonucu Doğrulama ve Callback Endpoint'i
app.post('/api/payment/callback', (req, res) => {
    const { token } = req.body;

    if (!token) {
        return res.status(400).json({ status: 'error', message: 'Geçersiz ödeme token parametresi.' });
    }

    const request = {
        locale: Iyzipay.LOCALE.TR,
        token: token
    };

    // İyzico'dan ödeme sonucunu güvenli şekilde sorgula
    iyzipay.checkoutForm.retrieve(request, (err, result) => {
        if (err || !result || result.status !== 'SUCCESS') {
            return res.redirect('/odeme-basarisiz?error=' + encodeURIComponent(result?.errorMessage || 'Ödeme doğrulama başarısız'));
        }

        // Ödeme durumunun tam olarak SUCCESS olduğunu doğrula
        if (result.paymentStatus === 'SUCCESS') {
            const orderId = result.conversationId;
            const paidAmount = result.paidPrice;
            
            // BURADA: Veritabanında ilgili siparişin durumu 'PAID' olarak güncellenir ve ödeme kaydı düşülür.
            // Örn: await db.update(orders).set({ status: 'PAID', paidAt: new Date() }).where(eq(orders.id, orderId));

            return res.redirect(`/odeme-basarili?orderId=${orderId}&amount=${paidAmount}`);
        } else {
            return res.redirect('/odeme-basarisiz?reason=payment_declined');
        }
    });
});

app.listen(PORT, () => {
    console.log(`MMT Connect Güvenli İyzico Ödeme Servisi ${PORT} portunda çalışmaktadır.`);
});
