// MMT Connect - Kesintisiz ve Gerçek DB Onaylı Lojistik Webhook Alıcısı

app.post('/api/webhook/logistics-inbound', async (req, res) => {
    try {
        const { customerName, origin, destination, cargoDetails, contactNumber, messageId } = req.body;

        // 1. Alan Doğrulama ve Güvenlik Kontrolü
        if (!customerName || !origin || !destination || !contactNumber || !messageId) {
            return res.status(400).json({ 
                success: false, 
                message: 'Eksik veya geçersiz parametre! Zorunlu alanlar doldurulmalıdır.' 
            });
        }

        // 2. Mükerrer (Tekrar Eden) Mesaj Kontrolü (Veritabanı Seviyesinde)
        const existingRequest = await db.findRequestByMessageId(messageId);
        if (existingRequest) {
            return res.status(200).json({ 
                success: true, 
                message: 'Bu talep daha önce sisteme işlenmiş (Mükerrer kayıt engellendi).' 
            });
        }

        // 3. Kalıcı Veritabanı Kaydı Hazırlığı
        const newRequestData = {
            id: 'MMT-' + Date.now(),
            messageId,
            customerName,
            origin, // Örn: Mersin / Gaziantep
            destination, // Örn: Habur / Zaxo / İbrahim Halil
            cargoDetails,
            contactNumber,
            status: 'PENDING_BROKER_MATCH',
            createdAt: new Date()
        };

        // Gerçek Veritabanına Kayıt İşlemi (Yorum satırından çıkarıldı ve aktif edildi)
        await db.saveLogisticsRequest(newRequestData);

        // 4. Sadece Gerçek Kayıt Başarılı Olduktan Sonra Yanıt Dönülür
        return res.status(200).json({ 
            success: true, 
            message: 'Talep MMT Connect veritabanına başarıyla kaydedildi.', 
            dataId: newRequestData.id 
        });

    } catch (error) {
        console.error('Webhook kritik hata:', error.message);
        return res.status(500).json({ 
            success: false, 
            message: 'Veritabanı kayıt sırasında hata oluştu.' 
        });
    }
});
