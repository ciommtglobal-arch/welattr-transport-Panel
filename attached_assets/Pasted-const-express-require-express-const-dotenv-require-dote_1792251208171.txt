const express = require('express');
const dotenv = require('dotenv');
const Iyzipay = require('iyzipay');

dotenv.config();

const app = express();
const PORT = process.env.PORT || 3000;

app.use(express.json());

// Iyzico İstemcisini Yapılandır (Canlı veya Test anahtarları .env'den okunur)
const iyzipay = new Iyzipay({
    apiKey: process.env.IYZICO_API_KEY,
    secretKey: process.env.IYZICO_SECRET_KEY,
    uri: process.env.IYZICO_BASE_URI || 'https://api.iyzipay.com' // Test için 'https://sandbox-api.iyzipay.com' kullanılabilir
});

// Sağlık Kontrolü
app.get('/', (req, res) => {
    res.json({
        status: 'success',
        message: 'MMT Connect - Canlı Iyzico Ödeme Servisi Aktif'
    });
});

// Gerçek Iyzico Ödeme Başlatma Endpoint'i
app.post('/api/payment/initialize', (req, res) => {
    const { price, paidPrice, basketId, buyer, shippingAddress, billingAddress, basketItems } = req.body;

    // Çelik Kubbe / Güvenlik ve Tutar Doğrulama Katmanı
    if (!price || !paidPrice || !buyer || !basketItems) {
        return res.status(400).json({
            status: 'error',
            message: 'Eksik ödeme parametreleri. Tutar, alıcı ve sepet bilgileri zorunludur.'
        });
    }

    const request = {
        locale: Iyzipay.LOCALE.TR,
        conversationId: basketId || 'MMT-' + Date.now(),
        price: price.toString(),
        paidPrice: paidPrice.toString(),
        currency: Iyzipay.CURRENCY.TRY,
        basketId: basketId || 'B98765',
        paymentGroup: Iyzipay.PAYMENT_GROUP.PRODUCT,
        buyer: {
            id: buyer.id || 'BY789',
            name: buyer.name,
            surname: buyer.surname,
            gsmNumber: buyer.gsmNumber,
            email: buyer.email,
            identityNumber: buyer.identityNumber || '11111111111',
            lastLoginDate: '2026-10-17 18:32:00',
            registrationDate: buyer.registrationDate || '2026-01-01 12:00:00',
            registrationAddress: buyer.registrationAddress,
            ip: req.ip || '85.105.0.1',
            city: buyer.city || 'Istanbul',
            country: buyer.country || 'Turkey',
            zipCode: buyer.zipCode || '34000'
        },
        shippingAddress: shippingAddress || {
            contactName: `${buyer.name}${buyer.surname}`,
            city: 'Istanbul',
            country: 'Turkey',
            address: 'MMT Global Lojistik Merkez',
            zipCode: '34000'
        },
        billingAddress: billingAddress || {
            contactName: `${buyer.name} `${buyer.surname}`,
            city: 'Istanbul',
            country: 'Turkey',
            address: 'MMT Global Lojistik Merkez',
            zipCode: '34000'
        },
        basketItems: basketItems.map((item, index) => ({
            id: item.id || `BI${index + 1}`,
            name: item.name,
            category1: item.category1 || 'Lojistik / Ticaret',
            itemType: Iyzipay.BASKET_ITEM_TYPE.PHYSICAL,
            price: item.price.toString()
        }))
    };

    // Iyzico Checkout Form Initialize Servisini Çağır
    iyzipay.checkoutFormInitialize.create(request, (err, result) => {
        if (err) {
            return res.status(500).json({
                status: 'error',
                message: 'Iyzico bağlantı hatası',
                error: err.message
            });
        }

        res.json({
            status: result.status, // SUCCESS veya FAILURE
            errorCode: result.errorCode,
            errorMessage: result.errorMessage,
            checkoutFormContent: result.paymentPageUrl, // Ödeme sayfasının HTML içeriği veya URL'si
            token: result.token
        });
    });
});

app.listen(PORT, () => {
    console.log(`MMT Connect Iyzico Servisi ${PORT} portunda canlı olarak çalışıyor.`);
});
