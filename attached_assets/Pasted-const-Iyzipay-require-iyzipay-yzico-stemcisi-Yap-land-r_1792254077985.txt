const Iyzipay = require('iyzipay');

// İyzico İstemcisi Yapılandırması
const iyzipay = new Iyzipay({
    apiKey: process.env.IYZICO_API_KEY,
    secretKey: process.env.IYZICO_SECRET_KEY,
    uri: process.env.IYZICO_BASE_URI || 'https://sandbox-api.iyzipay.com'
});

function registerPaymentRoutes(app) {
    // Doğrudan çalışan ödeme başlatma rotası
    app.post('/api/payment/initialize', (req, res) => {
        const { price, basketId, buyer, basketItems } = req.body;

        const request = {
            locale: Iyzipay.LOCALE.TR,
            conversationId: basketId || 'MMT-' + Date.now(),
            price: price ? price.toString() : '100.00',
            paidPrice: price ? price.toString() : '100.00',
            currency: Iyzipay.CURRENCY.TRY,
            basketId: basketId || 'BASKET-1',
            paymentGroup: Iyzipay.PAYMENT_GROUP.PRODUCT,
            callbackUrl: `${process.env.APP_BASE_URL || 'https://panel.mmtglobal.com.tr'}/api/payment/callback`,
            buyer: {
                id: buyer?.id || 'USER-1',
                name: buyer?.name || 'Mehmet',
                surname: buyer?.surname || 'Alioğlu',
                gsmNumber: buyer?.gsmNumber || '+905000000000',
                email: buyer?.email || 'cio@mmtglobal.com.tr',
                identityNumber: '11111111111',
                lastLoginDate: '2026-10-17 19:25:00',
                registrationDate: '2026-01-01 12:00:00',
                registrationAddress: 'MMT Global Merkez Cizre',
                ip: req.ip || '85.105.0.1',
                city: 'Sirnak',
                country: 'Turkey',
                zipCode: '73200'
            },
            shippingAddress: {
                contactName: 'Mehmet Alioğlu',
                city: 'Sirnak',
                country: 'Turkey',
                address: 'MMT Global Merkez Cizre',
                zipCode: '73200'
            },
            billingAddress: {
                contactName: 'Mehmet Alioğlu',
                city: 'Sirnak',
                country: 'Turkey',
                address: 'MMT Global Merkez Cizre',
                zipCode: '73200'
            },
            basketItems: basketItems || [
                {
                    id: 'ITEM-1',
                    name: 'MMT Lojistik Hizmet Bedeli',
                    category1: 'Lojistik',
                    itemType: Iyzipay.BASKET_ITEM_TYPE.PHYSICAL,
                    price: price ? price.toString() : '100.00'
                }
            ]
        };

        iyzipay.checkoutFormInitialize.create(request, (err, result) => {
            if (err) {
                return res.status(500).json({ status: 'error', message: err.message });
            }
            res.json(result);
        });
    });

    // Ödeme dönüş callback rotası
    app.post('/api/payment/callback', (req, res) => {
        const { token } = req.body;
        if (!token) {
            return res.status(400).json({ status: 'error', message: 'Token eksik' });
        }

        iyzipay.checkoutForm.retrieve({ locale: Iyzipay.LOCALE.TR, token }, (err, result) => {
            if (err || result.status !== 'success') {
                return res.redirect('/odeme-basarisiz');
            }
            if (result.paymentStatus === 'SUCCESS') {
                return res.redirect(`/odeme-basarili?orderId=${result.conversationId}`);
            } else {
                return res.redirect('/odeme-basarisiz');
            }
        });
    });
}

module.exports = registerPaymentRoutes;
