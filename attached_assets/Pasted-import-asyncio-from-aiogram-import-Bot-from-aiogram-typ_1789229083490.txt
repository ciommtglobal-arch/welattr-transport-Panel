import asyncio
from aiogram import Bot
from aiogram.types import InlineKeyboardButton, InlineKeyboardMarkup

# Bot Token ve VIP Kanal ID
BOT_TOKEN = "BOT_TOKEN_BURAYA"
VIP_CHANNEL_ID = -1004409778193

bot = Bot(token=BOT_TOKEN)


async def send_vip_cargo(cargo: dict):
    # Mesaj metni şablonu (HTML formatı)
    text = (
        "🚛 <b>YENİ VIP YÜK BİLDİRİMİ</b>\n"
        "━━━━━━━━━━━━━━━━━━━━━\n"
        f"📍 <b>Çıkış:</b> {cargo.get('origin')}\n"
        f"🏁 <b>Varış:</b> {cargo.get('destination')}\n"
        f"📦 <b>Yük Tipi:</b> {cargo.get('cargo_type')}\n"
        f"⚖️ <b>Ağırlık / Hacim:</b> {cargo.get('weight')}\n"
        f"🚚 <b>Dorse / Araç:</b> {cargo.get('truck_type')}\n"
        f"💵 <b>Navlun:</b> {cargo.get('price', 'Görüşülecek')}\n"
        f"📅 <b>Yükleme Tarihi:</b> {cargo.get('date')}\n"
        "━━━━━━━━━━━━━━━━━━━━━\n"
        f"ℹ️ <i>{cargo.get('note', 'Detaylar için yetkili ile iletişime geçin.')}</i>"
    )

    # Direkt teklif/iletişim butonu
    keyboard = InlineKeyboardMarkup(
        inline_keyboard=[
            [
                InlineKeyboardButton(
                    text="📞 Yükü Al / İletişime Geç",
                    url=f"https://t.me/{cargo.get('contact_username', 'DestekBot')}",
                )
            ]
        ]
    )

    # Kanala gönderim
    await bot.send_message(
        chat_id=VIP_CHANNEL_ID,
        text=text,
        parse_mode="HTML",
        reply_markup=keyboard,
    )
