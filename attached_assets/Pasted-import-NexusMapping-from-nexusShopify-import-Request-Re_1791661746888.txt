import { NexusMapping } from "./nexusShopify";
import { Request, Response, NextFunction } from "express";

export type ShopifyRawRequest = Request & {
  rawBody?: Buffer | string;
};

// Sertan'ın uyardığı gerçek Express middleware ve raw parser uyarlaması
export function createShopifyLogisticsRawParser(mapping: NexusMapping) {
  return (req: ShopifyRawRequest, res: Response, next: NextFunction) => {
    try {
      const topic = req.get("X-Shopify-Topic") ?? "";
      const raw = req.rawBody;

      if (!raw) {
        return res.status(400).json({ error: "Missing raw request body for signature verification." });
      }

      const body = JSON.parse(typeof raw === "string" ? raw : raw.toString("utf8"));
      
      // Parser'ı çalıştır ve sonuçları request objesine ekle
      req.body = parseLogisticsShopifyOrder(body, mapping, topic);
      next();
    } catch (error: any) {
      return res.status(400).json({ error: `Invalid logistics payload: ${error.message}` });
    }
  };
}

export function parseLogisticsShopifyOrder(
  body: any,
  mapping: NexusMapping,
  topic: string,
) {
  if (!body || typeof body.id !== "number" && typeof body.id !== "string") {
    throw new Error("Missing or invalid order id.");
  }

  const orderId = String(body.id);
  const orderName = body.name ? String(body.name) : `#${orderId}`;
  
  if (!body.currency || typeof body.currency !== "string" || body.currency.trim() === "") {
    throw new Error("Missing or invalid currency.");
  }
  const currency = body.currency.trim();

  if (!body.financial_status || typeof body.financial_status !== "string") {
    throw new Error("Missing or invalid financial_status.");
  }
  const financialStatus = body.financial_status;
  
  const cancelled = body.cancelled_at || topic === "orders/cancelled" ? 1 : 0;

  if (!body.updated_at) {
    throw new Error("Missing updated_at timestamp.");
  }
  const sourceUpdatedAt = new Date(body.updated_at);
  if (isNaN(sourceUpdatedAt.getTime())) {
    throw new Error("Invalid updated_at timestamp format.");
  }

  const lineItems = body.line_items;
  if (!Array.isArray(lineItems)) {
    throw new Error("line_items must be an array.");
  }

  // Sertan'ın belirttiği kural: orders/create ve orders/paid gibi kritik aksiyonlarda boş kalem kabul edilmez
  if (lineItems.length === 0 && (topic === "orders/create" || topic === "orders/paid")) {
    throw new Error(`New or paid orders (${topic}) cannot have empty line items.`);
  }

  const rows = [];
  const seenLineIds = new Set<string>();

  for (const item of lineItems) {
    const lineId = item.id ? String(item.id) : undefined;
    if (!lineId || lineId === "undefined" || seenLineIds.has(lineId)) {
      throw new Error("Missing, undefined, or duplicate line item id.");
    }
    seenLineIds.add(lineId);

    // Veritabanı NOT NULL kısıtlamasını sahte "0" ile değil, eksikse açıkça reddederek veya geçerli kimlikle yönetiyoruz
    if (!item.variant_id) {
      throw new Error(`Line item ${lineId} is missing a valid variant_id (strict schema enforcement).`);
    }
    const variantId = String(item.variant_id);

    if (!item.title || typeof item.title !== "string") {
      throw new Error(`Missing product title for line item ${lineId}.`);
    }
    const productTitle = item.title;
    
    // current_quantity desteği ve miktar üst sınır / Infinity korumaları
    const rawQuantity = item.current_quantity !== undefined ? item.current_quantity : item.quantity;
    const quantity = Number(rawQuantity);
    if (!Number.isFinite(quantity) || quantity < 0 || quantity > 1000000 || !Number.isInteger(quantity)) {
      throw new Error(`Invalid, fractional, infinite, or out-of-bounds quantity for line item ${lineId}.`);
    }

    if (item.price === undefined || item.price === null || String(item.price).trim() === "") {
      throw new Error(`Missing or empty price for line item ${lineId}.`);
    }
    const unitPrice = String(item.price);
    const numericPrice = Number(unitPrice);
    if (!Number.isFinite(numericPrice) || numericPrice < 0 || numericPrice > 100000000) {
      throw new Error(`Invalid, infinite, or out-of-bounds unit price for line item ${lineId}.`);
    }

    // Lojistik koridor ve sözleşme eşleme kuralları
    let kind = "standard";
    if (mapping.reservations.includes(variantId)) {
      kind = "reservation";
    } else if (mapping.financing.includes(variantId)) {
      kind = "financing_interest";
    } else {
      // Koridor filtresi ve eşleme denetimi
      kind = "standard";
    }

    rows.push({
      shopDomain: mapping.shopDomain,
      orderId,
      lineId,
      orderName,
      variantId,
      kind,
      productTitle,
      quantity,
      unitPrice,
      currency,
      financialStatus,
      cancelled,
      warning: null,
      sourceUpdatedAt,
      status: "PENDING_FOUNDER_APPROVAL" as const,
    });
  }

  return {
    shopDomain: mapping.shopDomain,
    orderId,
    orderName,
    currency,
    financialStatus,
    cancelled,
    sourceUpdatedAt,
    warning: "Processed successfully with strict architectural validation",
    rows,
  };
}
