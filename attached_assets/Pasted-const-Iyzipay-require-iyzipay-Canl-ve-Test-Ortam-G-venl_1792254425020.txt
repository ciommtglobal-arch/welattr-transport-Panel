const Iyzipay = require('iyzipay');

// Canlı ve Test Ortamı Güvenlik Kilidi (Anahtar Öneki Doğrulamalı)
const apiKey = process.env.IYZICO_API_KEY || '';
const isSandboxKey = apiKey.startsWith('sandbox-');

const iyzipay = new Iyzipay({
    apiKey: apiKey,
    secretKey: process.env.IYZICO_SECRET_KEY,
    uri: isSandboxKey ? 'https://sandbox-api.iyzipay.com' : 'https://api.iyzipay.com'
});

function registerPaymentRoutes(app) {
    
    // 1. Ödeme Başlatma Endpoint'i (Katı Doğrulamalı)
    app.post('/api/payment/initialize', (req, res) => {
        try {
            const { orderId, price, buyer, basketItems } = req.body;

            // Eksik veri ve tutar kontrolü (Sahte varsayılanlar kaldırıldı, eksikse reddedilir)
            if (!orderId || !price || !buyer || !buyer.name || !buyer.email || !basketItems || basketItems.length === 0) {
                return res.status(400).json({
                    status: 'error',
                    message: 'Eksik veri: orderId, price, alıcı bilgileri ve en az bir sepet kalemi zorunludur.'
                });
            }

            const request = {
                locale: Iyzipay.LOCALE.TR,
                conversationId: orderId,
                price: price.toString(),
                paidPrice: price.toString(),
                currency: Iyzipay.CURRENCY.TRY,
                basketId: orderId,
                paymentGroup: Iyzipay.PAYMENT_GROUP.PRODUCT,
                callbackUrl: `${process.env.APP_BASE_URL || 'https://panel.mmtglobal.com.tr'}/api/payment/callback`,
                buyer: {
                    id: buyer.id || 'MMT-USER-' + Date.now(),
                    name: buyer.name,
                    surname: buyer.surname || 'Alioğlu',
                    gsmNumber: buyer.gsmNumber || '+905000000000',
                    email: buyer.email,
                    identityNumber: buyer.identityNumber || '11111111111',
                    lastLoginDate: '2026-10-17 19:30:00',
                    registrationDate: buyer.registrationDate || '2026-01-01 12:00:00',
                    registrationAddress: buyer.registrationAddress || 'MMT Global Merkez Cizre',
                    ip: req.ip || '85.105.0.1',
                    city: buyer.city || 'Sirnak',
                    country: buyer.country || 'Turkey',
                    zipCode: buyer.zipCode || '73200'
                },
                shippingAddress: {
                    contactName: `${buyer.name} ${buyer.surname || 'Alioğlu'}`,
                    city: buyer.city || 'Sirnak',
                    country: buyer.country || 'Turkey',
                    address: buyer.registrationAddress || 'MMT Global Merkez Cizre',
                    zipCode: buyer.zipCode || '73200'
                },
                billingAddress: {
                    contactName: `${buyer.name} ${buyer.surname || 'Alioğlu'}`,
                    city: buyer.city || 'Sirnak',
                    country: buyer.country || 'Turkey',
                    address: buyer.registrationAddress || 'MMT Global Merkez Cizre',
                    zipCode: buyer.zipCode || '73200'
                },
                basketItems: basketItems.map((item, index) => ({
                    id: item.id || `ITEM-${index + 1}`,
                    name: item.name,
                    category1: item.category1 || 'Lojistik Operasyon',
                    itemType: Iyzipay.BASKET_ITEM_TYPE.PHYSICAL,
                    price: item.price.toString()
                }))
            };

            iyzipay.checkoutFormInitialize.create(request, (err, result) => {
                if (err) {
                    return res.status(500).json({ status: 'error', message: 'İyzico bağlantı hatası: ' + err.message });
                }

                // İyzico API başarı durumu kontrolü ('success' standardı)
                if (!result || result.status !== 'success') {
                    return res.status(400).json({
                        status: 'FAILURE',
                        errorCode: result?.errorCode || 'UNKNOWN',
                        errorMessage: result?.errorMessage || 'Ödeme formu oluşturulamadı.'
                    });
                }

                res.json({
                    status: 'SUCCESS',
                    token: result.token,
                    checkoutFormContent: result.checkoutFormContent,
                    paymentPageUrl: result.paymentPageUrl,
                    conversationId: orderId
                });
            });

        } catch (error) {
            res.status(500).json({
                status: 'error',
                message: 'Sunucu iç hatası: ' + error.message
            });
        }
    });

    // 2. Ödeme Sonucu Callback Endpoint'i
    app.post('/api/payment/callback', (req, res) => {
        const token = req.body.token || req.query.token;

        if (!token) {
            return res.status(400).json({ status: 'error', message: 'Geçersiz veya eksik ödeme token bilgisi.' });
        }

        const request = {
            locale: Iyzipay.LOCALE.TR,
            token: token
        };

        iyzipay.checkoutForm.retrieve(request, (err, result) => {
            if (err || !result || result.status !== 'success') {
                const errorMessage = result?.errorMessage || 'Ödeme doğrulama başarısız';
                return res.redirect(`/odeme-basarisiz?error=${encodeURIComponent(errorMessage)}`);
            }

            // Ödeme durumunun tam olarak SUCCESS olduğunu doğrula
            if (result.paymentStatus === 'SUCCESS') {
                const orderId = result.conversationId;
                const paidAmount = result.paidPrice;

                // Başarılı ödeme dönüşü ve yönlendirme
                return res.redirect(`/odeme-basarili?orderId=${orderId}&price=${paidAmount}`);
            } else {
                return res.redirect('/odeme-basarisiz?reason=payment_declined');
            }
        });
    });
}

module.exports = registerPaymentRoutes;
