 * MMT Global & Shopify Gerçek Veritabanı Entegrasyon Modülü
 * Koridor: Samsun - Karadeniz - Irak & Gürcistan Boru Hattı
 * Güvenlik: Çelik Kubbe & WelatTR (Gerçek DB Kalıcılığı ve Güvenli HMAC)
 */

const crypto = require('crypto');
const express = require('express');
const router = express.Router();

// Gerçek Veritabanı Bağlantısı (Örn: MongoDB / Mongoose veya PostgreSQL modeli)
const ApprovalQueueModel = require('../models/ApprovalQueue'); 

/**
 * 1. Adım: Güvenli Zaman Sabitli HMAC İmza Doğrulama
 */
function verifyShopifyWebhook(req, res, next) {
    const hmacHeader = req.get('X-Shopify-Hmac-Sha256');
    const secret = process.env.SHOPIFY_WEBHOOK_SECRET;

    if (!hmacHeader || !secret) {
        return res.status(401).json({ 
            status: 'error', 
            message: 'Çelik Kubbe: İmza veya gizli anahtar eksik.' 
        });
    }

    const calculatedHmac = crypto
        .createHmac('sha256', secret)
        .update(JSON.stringify(req.body))
        .digest('base64');

    // Güvenlik zafiyetini önlemek için constant-time comparison (zaman sabitli karşılaştırma)
    const bufferA = Buffer.from(calculatedHmac);
    const bufferB = Buffer.from(hmacHeader);

    if (bufferA.length !== bufferB.length || !crypto.timingSafeEqual(bufferA, bufferB)) {
        return res.status(403).json({ 
            status: 'error', 
            message: 'Çelik Kubbe: Geçersiz imza, erişim reddedildi.' 
        });
    }

    next();
}

/**
 * 2. Adım: Veritabanına Kalıcı Kayıt Ekleyen Güvenli Endpoint
 */
router.post('/api/mmt/secure-sync', verifyShopifyWebhook, async (req, res) => {
    try {
        const { product_id, title, inventory_quantity, corridor, destination } = req.body;

        const targetCorridor = corridor || 'Samsun-Irak-Proje';
        if (!['Samsun-Irak-Proje', 'Samsun-Gurcistan'].includes(targetCorridor)) {
            return res.status(400).json({ 
                status: 'error', 
                message: 'Çelik Kubbe: Geçersiz lojistik koridoru.' 
            });
        }

        // Gerçek Veritabanı Modeline Kayıt Verisi Hazırlama
        const newQueueItem = new ApprovalQueueModel({
            transactionId: `MMT-SECURE-DB-${Date.now()}`,
            productId: product_id || 'N/A',
            itemTitle: title || 'Proje / Boru Sevkiyat Kalemi',
            stockCount: inventory_quantity || 0,
            corridor: targetCorridor,
            destination: destination || 'Irak Şantiye Sahası',
            approvalStatus: 'Kurucu Onayı Bekliyor',
            authorizedBy: 'WelatTR (Beklemede)',
            createdAt: new Date()
        });

        // Veritabanına kalıcı olarak yazma işlemi
        await newQueueItem.save();

        console.log(`[MMT Nexus DB] Başarıyla veritabanına işlendi ve onay kuyruğuna eklendi: ${title}`);

        return res.status(200).json({
            status: 'success',
            message: 'Veritabanı kalıcılığı sağlandı. Talep Kurucu Onay Kuyruğuna eklendi.',
            data: newQueueItem
        });

    } catch (error) {
        console.error('Veritabanı Entegrasyon Hatası:', error.message);
        return res.status(500).json({ 
            status: 'error', 
            message: 'Veritabanı kayıt hatası: ' + error.message 
        });
    }
});

module.exports = router;
