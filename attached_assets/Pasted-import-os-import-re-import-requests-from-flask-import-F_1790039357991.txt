import os
import re
import requests
from flask import Flask, request, jsonify

app = Flask(__name__)

# Replit Secrets (Ortam Değişkenleri) üzerinden okunur
TELEGRAM_BOT_TOKEN = os.environ.get("TELEGRAM_BOT_TOKEN")
TELEGRAM_CHAT_ID = os.environ.get("TELEGRAM_CHAT_ID")

def clean_phone_number(phone_raw):
    """Telefon numarasından +, boşluk, tire vb. kaldırır, 90 standardına çeker."""
    if not phone_raw:
        return ""
    # Sadece rakamları al
    digits = re.sub(r"\D", "", phone_raw)
    
    # 05xx girildiyse 905xx formatına çevir
    if digits.startswith("0") and len(digits) == 11:
        digits = "90" + digits[1:]
    # 5xx (10 haneli) girildiyse başına 90 ekle
    elif len(digits) == 10 and digits.startswith("5"):
        digits = "90" + digits
        
    return digits

def send_telegram_alert(order_data):
    """Telegram grubuna sipariş bildirimi ve butonları gönderir."""
    order_id = order_data.get("name", str(order_data.get("id", "")))
    total_price = order_data.get("total_price", "0.00")
    currency = order_data.get("currency", "TRY")
    
    # Müşteri Bilgileri
    customer = order_data.get("customer", {})
    first_name = customer.get("first_name", "")
    last_name = customer.get("last_name", "")
    customer_name = f"{first_name} {last_name}".strip() or "Belirtilmedi"
    
    # Telefon alma (müşteri veya teslimat adresinden)
    phone_raw = customer.get("phone") or order_data.get("shipping_address", {}).get("phone", "")
    phone_clean = clean_phone_number(phone_raw)
    
    # Ürün Detayları
    items = order_data.get("line_items", [])
    item_titles = ", ".join([f"{item.get('title')} (x{item.get('quantity')})" for item in items]) or "Ürün bilgisi yok"
    
    # Telegram Mesaj Metni
    message_text = (
        f"🚨 <b>YENİ SİPARİŞ / KAPORA ALINDI!</b>\n\n"
        f"• <b>Sipariş No:</b> {order_id}\n"
        f"• <b>Müşteri:</b> {customer_name}\n"
        f"• <b>Telefon:</b> {phone_raw or 'Kayıtlı değil'}\n"
        f"• <b>Ürün / Hizmet:</b> {item_titles}\n"
        f"• <b>Ödenen Tutar:</b> {total_price} {currency}\n"
        f"• <b>Durum:</b> Ödeme Onaylandı (PayTR/Shopify)\n"
    )
    
    # Butonlar (Inline Keyboard)
    inline_keyboard = []
    
    if phone_clean:
        inline_keyboard.append([
            {"text": "💬 Müşteriyle WhatsApp'ta Görüş", "url": f"https://wa.me/{phone_clean}"}
        ])
        inline_keyboard.append([
            {"text": "📞 Müşteriyi Doğrudan Ara", "url": f"tel:{phone_clean}"}
        ])
    
    payload = {
        "chat_id": TELEGRAM_CHAT_ID,
        "text": message_text,
        "parse_mode": "HTML",
        "reply_markup": {"inline_keyboard": inline_keyboard} if inline_keyboard else {}
    }
    
    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
    response = requests.post(url, json=payload)
    return response.json()

@app.route("/", methods=["GET"])
def home():
    return "MMT Global Shopify-Telegram Webhook Sunucusu Aktif.", 200

@app.route("/webhook/shopify/order-paid", methods=["POST"])
def shopify_order_paid():
    """Shopify 'Order payment' webhook'unu karşılayan uç nokta."""
    data = request.get_json(silent=True)
    if not data:
        return jsonify({"status": "error", "message": "Geçersiz veri"}), 400
    
    # Yalnızca ödemesi tamamlanmış siparişleri Telegram'a at
    financial_status = data.get("financial_status")
    if financial_status == "paid":
        send_telegram_alert(data)
        return jsonify({"status": "success", "message": "Telegram bildirimi gönderildi"}), 200
    
    return jsonify({"status": "ignored", "message": f"Sipariş ödenmedi ({financial_status})"}), 200

if __name__ == "__main__":
    # Replit için 0.0.0.0 ve 8080 portu
    app.run(host="0.0.0.0", port=8080)
