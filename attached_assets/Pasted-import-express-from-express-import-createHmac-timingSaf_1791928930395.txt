import express from "express";
import { createHmac, timingSafeEqual } from "node:crypto";
import { Pool } from "pg";

const SECRET_KEY = process.env.WELATTR_PRODUCTION_SECURE_KEY;
const DATABASE_URL = process.env.DATABASE_URL;

if (!SECRET_KEY || !DATABASE_URL) {
  console.error("Replit Secrets içinde DATABASE_URL ve WELATTR_PRODUCTION_SECURE_KEY tanımlanmalı.");
  process.exit(1);
}

const app = express();
const pool = new Pool({ connectionString: DATABASE_URL });

const ALLOWED_KINDS = new Set([
  "invoice_verification",
  "identity_document",
  "contract_review",
  "compliance_check"
]);

app.use(express.json({
  limit: "16kb",
  verify(req, _res, buffer) {
    req.rawBody = buffer;
  }
}));

async function initDb() {
  await pool.query(`
    CREATE TABLE IF NOT EXISTS mmt_events (
      event_id VARCHAR(128) PRIMARY KEY,
      ts BIGINT NOT NULL,
      doc_ref VARCHAR(256) NOT NULL,
      kind VARCHAR(64) NOT NULL,
      status VARCHAR(32) NOT NULL,
      created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );

    CREATE TABLE IF NOT EXISTS review_queue (
      event_id VARCHAR(128) PRIMARY KEY
        REFERENCES mmt_events(event_id),
      status VARCHAR(32) NOT NULL,
      created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );

    CREATE TABLE IF NOT EXISTS audit_logs (
      id BIGSERIAL PRIMARY KEY,
      event_id VARCHAR(128) NOT NULL
        REFERENCES mmt_events(event_id),
      action VARCHAR(64) NOT NULL,
      actor VARCHAR(128) NOT NULL,
      created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );
  `);
}

function verifyMmt(req, res, next) {
  const signature = req.get("X-MMT-Signature");
  const timestamp = req.get("X-MMT-Timestamp");
  const eventId = req.get("X-MMT-Event-Id");

  if (
    !signature ||
    !/^[a-f0-9]{64}$/i.test(signature) ||
    !timestamp ||
    !/^\d{13}$/.test(timestamp) ||
    !eventId ||
    !/^[a-zA-Z0-9_-]{1,128}$/.test(eventId) ||
    !req.rawBody
  ) {
    return res.status(401).json({ error: "İstek doğrulanamadı." });
  }

  const requestTime = Number(timestamp);
  if (Math.abs(Date.now() - requestTime) > 5 * 60 * 1000) {
    return res.status(401).json({ error: "İstek zaman aşımına uğradı." });
  }

  const signedData = Buffer.concat([
    Buffer.from(`${timestamp}.${eventId}.`, "utf8"),
    req.rawBody
  ]);

  const expected = createHmac("sha256", SECRET_KEY)
    .update(signedData)
    .digest();

  const received = Buffer.from(signature, "hex");

  if (
    received.length !== expected.length ||
    !timingSafeEqual(received, expected)
  ) {
    return res.status(401).json({ error: "İstek doğrulanamadı." });
  }

  req.mmt = { id: eventId, ts: requestTime };
  next();
}

app.get("/health", async (_req, res) => {
  try {
    await pool.query("SELECT 1");
    res.json({ ok: true, database: "connected" });
  } catch {
    res.status(503).json({ ok: false, database: "unavailable" });
  }
});

app.post("/events", verifyMmt, async (req, res) => {
  const { docRef, kind } = req.body ?? {};

  if (
    typeof docRef !== "string" ||
    !/^[a-zA-Z0-9_:-]{1,256}$/.test(docRef) ||
    typeof kind !== "string" ||
    !ALLOWED_KINDS.has(kind)
  ) {
    return res.status(400).json({ error: "Geçersiz docRef veya kind." });
  }

  const { id, ts } = req.mmt;
  let client;
  let inTransaction = false;

  try {
    client = await pool.connect();
    await client.query("BEGIN");
    inTransaction = true;

    const inserted = await client.query(
      `INSERT INTO mmt_events (event_id, ts, doc_ref, kind, status)
       VALUES ($1, $2, $3, $4, 'queued')
       ON CONFLICT (event_id) DO NOTHING
       RETURNING event_id`,
      [id, ts, docRef, kind]
    );

    if (inserted.rowCount === 0) {
      const result = await client.query(
        `SELECT
           e.doc_ref,
           e.kind,
           EXISTS (
             SELECT 1 FROM review_queue q
             WHERE q.event_id = e.event_id AND q.status = 'pending'
           ) AS queued,
           EXISTS (
             SELECT 1 FROM audit_logs a
             WHERE a.event_id = e.event_id
           ) AS audited
         FROM mmt_events e
         WHERE e.event_id = $1`,
        [id]
      );

      const existing = result.rows[0];

      if (
        !existing ||
        existing.doc_ref !== docRef ||
        existing.kind !== kind
      ) {
        await client.query("ROLLBACK");
        inTransaction = false;
        return res.status(409).json({
          error: "Bu event_id farklı içerikle daha önce kullanılmış."
        });
      }

      await client.query("COMMIT");
      inTransaction = false;

      return res.status(200).json({
        ok: true,
        eventId: id,
        alreadyReceived: true,
        persisted: true,
        queuedForReview: existing.queued,
        auditLogged: existing.audited
      });
    }

    await client.query(
      `INSERT INTO review_queue (event_id, status)
       VALUES ($1, 'pending')`,
      [id]
    );

    await client.query(
      `INSERT INTO audit_logs (event_id, action, actor)
       VALUES ($1, 'QUEUED', 'mmt_hmac_peer')`,
      [id]
    );

    await client.query("COMMIT");
    inTransaction = false;

    return res.status(200).json({
      ok: true,
      eventId: id,
      alreadyReceived: false,
      persisted: true,
      queuedForReview: true,
      auditLogged: true,
      shipmentApproved: false
    });
  } catch (error) {
    if (client && inTransaction) {
      await client.query("ROLLBACK").catch(() => {});
    }

    console.error("Event kaydı başarısız.");
    return res.status(500).json({ error: "Veritabanı işlemi başarısız." });
  } finally {
    client?.release();
  }
});

app.post("/approve", (_req, res) => {
  res.status(501).json({
    error: "Onay işlemi kullanıcı doğrulaması kurulana kadar kapalı."
  });
});

async function start() {
  try {
    await initDb();

    const port = Number(process.env.PORT) || 3000;
    app.listen(port, "0.0.0.0", () => {
      console.log(`MMT Connect API ${port} portunda çalışıyor.`);
    });
  } catch {
    console.error("Veritabanı başlatılamadı; servis açılmadı.");
    process.exit(1);
  }
}

start();
