import { Router, Request, Response } from "express";
import { and, eq, lte, notInArray, sql } from "drizzle-orm";
import {
  db,
  nexusShopifyEventsTable as events,
  nexusShopifyIntakesTable as intakes,
} from "@workspace/db";

  nexusMapping,

  type ShopifyRawRequest,

const router = Router();

const topics = new Set([
  "orders/create",
  "orders/paid",
  "orders/updated",
  "orders/cancelled",
]);

const pending = "PENDING_FOUNDER_APPROVAL" as const;

type SourceRow =
  ReturnType<typeof parseLogisticsShopifyOrder>["rows"][number];

const sourceFields = [
  "orderName",
  "variantId",
  "kind",
  "productTitle",
  "quantity",
  "unitPrice",
  "currency",
  "financialStatus",
  "cancelled",
  "warning",
] as const;

function sameSource(
  old: typeof intakes.$inferSelect,
  row: SourceRow,
): boolean {
  return sourceFields.every(field => old[field] === row[field]);
}

// İçerik değişmediyse kurucunun kararını korur, değiştiyse tekrar PENDING durumuna çeker.
function nextStatus(row: SourceRow) {
  const changed = sql.join(
    sourceFields.map(
      field => sql`${intakes[field]} IS DISTINCT FROM ${row[field]}`,
    ),
    sql` OR `,
  );

  return sql`
    CASE
      WHEN (${changed})
        THEN ${pending}::nexus_approval_status
      ELSE ${intakes.status}
    END
  `;
}

class ConflictingSnapshot extends Error {}

router.post("/mmt/secure-sync", async (req: Request, res: Response) => {
  const mapping = nexusMapping(true);
  const secret = process.env.SHOPIFY_NEXUS_WEBHOOK_SECRET;

  if (!mapping || !secret) {
    res.status(503).json({
      error: "Shopify logistics configuration is incomplete.",
    });
    return;
  }

  const raw = (req as ShopifyRawRequest).rawBody;

  // Güvenli ham gövde (raw body) ve imza kontrolü
  if (
    (typeof raw !== "string" && !Buffer.isBuffer(raw)) ||
    !verifyShopifySignature(
      raw,
      req.get("X-Shopify-Hmac-Sha256") ?? "",
      secret,
    )
  ) {
    res.status(401).json({
      error: "Missing raw body or invalid Shopify signature.",
    });
    return;
  }

  if (
    req.get("X-Shopify-Shop-Domain")?.toLowerCase() !==
    mapping.shopDomain
  ) {
    res.status(403).json({
      error: "Unexpected Shopify shop.",
    });
    return;
  }

  const topic = req.get("X-Shopify-Topic") ?? "";
  const webhookId = req.get("X-Shopify-Webhook-Id") ?? "";

  if (
    !topics.has(topic) ||
    !/^[a-zA-Z0-9-]{1,128}$/.test(webhookId)
  ) {
    res.status(400).json({
      error: "Invalid Shopify topic or webhook identity.",
    });
    return;
  }

  let snapshot: ReturnType<typeof parseLogisticsShopifyOrder>;

  try {
    // HMAC güvenliği doğrulandıktan sonra güvenli ayrıştırma
    snapshot = parseLogisticsShopifyOrder(
      JSON.parse(
        typeof raw === "string" ? raw : raw.toString("utf8"),
      ),
      mapping,
      topic,
    );
  } catch {
    res.status(400).json({
      error:
        "Invalid order data, required line fields or logistics corridor.",
    });
    return;
  }

  try {
    const result = await db.transaction(async tx => {
      // Eşzamanlı yarış durumlarını (race condition) önlemek için transaction kilidi
      await tx.execute(sql`
        SELECT pg_advisory_xact_lock(
          hashtextextended(
            ${`${mapping.shopDomain}:${snapshot.orderId}`},
            0
          )
        )
      `);

      const [event] = await tx
        .insert(events)
        .values({
          shopDomain: mapping.shopDomain,
          webhookId: `logistics:${webhookId}`,
          topic,
        })
        .onConflictDoNothing()
        .returning({ id: events.id });

      if (!event) {
        return {
          duplicate: true,
          stale: false,
          received: 0,
          removed: 0,
        };
      }

      const orderWhere = and(
        eq(intakes.shopDomain, mapping.shopDomain),
        eq(intakes.orderId, snapshot.orderId),
      );

      const previous = await tx
        .select()
        .from(intakes)
        .where(orderWhere);

      const latest = Math.max(
        ...previous.map(row => row.sourceUpdatedAt.getTime()),
        -Infinity,
      );

      const incoming = snapshot.sourceUpdatedAt.getTime();

      if (incoming < latest) {
        return {
          duplicate: false,
          stale: true,
          received: 0,
          removed: 0,
        };
      }

      const byLine = new Map(
        previous.map(row => [row.lineId, row]),
      );

      const lineIds = snapshot.rows.map(row => row.lineId);

      const removedRows = previous.filter(
        row => !lineIds.includes(row.lineId),
      );

      const removalWarning =
        `${snapshot.warning}; Shopify siparişinde bu kalem kaldırıldı.`;

      const removalRows = removedRows.map(row => ({
        ...row,
        orderName: snapshot.orderName,
        quantity: 0,
        currency: snapshot.currency,
        financialStatus: snapshot.financialStatus,
        cancelled: snapshot.cancelled,
        warning: removalWarning,
        status: pending,
        sourceUpdatedAt: snapshot.sourceUpdatedAt,
      }));

      if (
        incoming === latest &&
        (
          snapshot.rows.some(
            row =>
              !byLine.has(row.lineId) ||
              !sameSource(byLine.get(row.lineId)!, row),
          ) ||
          removalRows.some(
            row => !sameSource(byLine.get(row.lineId)!, row),
          )
        )
      ) {
        throw new ConflictingSnapshot();
      }

      // Variant_id null değerlerini destekleyecek şekilde veritabanı kayıt döngüsü
      for (const row of snapshot.rows) {
        const {
          shopDomain: _shop,
          orderId: _order,
          lineId: _line,
          status: _status,
          ...source
        } = row;

        await tx
          .insert(intakes)
          .values(row)
          .onConflictDoUpdate({
            target: [
              intakes.shopDomain,
              intakes.orderId,
              intakes.lineId,
            ],
            set: {
              ...source,
              status: nextStatus(row),
            },
            setWhere: lte(
              intakes.sourceUpdatedAt,
              snapshot.sourceUpdatedAt,
            ),
          });
      }

      // Kaldırılan kalemlerin miktarlarını 0 olarak güncelleyip loglama
      for (const row of removalRows) {
        await tx
          .update(intakes)
          .set({
            orderName: row.orderName,
            quantity: 0,
            currency: row.currency,
            financialStatus: row.financialStatus,
            cancelled: row.cancelled,
            warning: row.warning,
            sourceUpdatedAt: row.sourceUpdatedAt,
            status: nextStatus(row),
          })
          .where(
            and(
              orderWhere,
              eq(intakes.lineId, row.lineId),
              lte(
                intakes.sourceUpdatedAt,
                snapshot.sourceUpdatedAt,
              ),
              lineIds.length
                ? notInArray(intakes.lineId, lineIds)
                : undefined,
            ),
          );
      }

      return {
        duplicate: false,
        stale: false,
        received: snapshot.rows.length,
        removed: removalRows.length,
      };
    });
