// MMT Connect - Güvenli WhatsApp / Lojistik Webhook Alıcısı
// Not: Ayrı bir app.listen açılmaz, mevcut ana API sunucusuna bağlanır.

app.post('/api/webhook/logistics-inbound', async (req, res) => {
    try {
        const { customerName, origin, destination, cargoDetails, contactNumber, messageId } = req.body;

        // 1. Veri Güvenliği ve Zorunlu Alan Kontrolü
        if (!customerName || !origin || !destination || !contactNumber) {
            return res.status(400).json({ success: false, message: 'Eksik parametre! Zorunlu alanlar doldurulmalıdır.' });
        }

        // 2. Tekrar Kontrolü (Aynı mesaj ID mükerrer kayıt oluşturmasın)
        // const existingRequest = await db.findRequestByMessageId(messageId);
        // if (existingRequest) { return res.status(200).json({ success: true, message: 'Talep zaten kayıtlı.' }); }

        // 3. Kalıcı Veritabanı Kaydı (MMT Connect Veritabanı)
        const newRequestData = {
            id: 'MMT-' + Date.now(),
            messageId: messageId || 'MANUAL-' + Date.now(),
            customerName,
            origin, // Örn: Mersin / Gaziantep
            destination, // Örn: Habur / Zaxo / İbrahim Halil
            cargoDetails,
            contactNumber,
            status: 'PENDING_BROKER_MATCH',
            createdAt: new Date()
        };

        // Veritabanına kayıt simülasyonu (Gerçek DB fonksiyonu buraya bağlanacak)
        // await db.saveLogisticsRequest(newRequestData);

        console.log('Yeni lojistik talebi MMT Connect veritabanına güvenle kaydedildi.');

        // 4. Başarı Yanıtı (Yalnızca kalıcı kayıt tamamlandıktan sonra döner)
        return res.status(200).json({ 
            success: true, 
            message: 'Talep MMT Connect paneline başarıyla işlendi.', 
            dataId: newRequestData.id 
        });

    } catch (error) {
        console.error('Webhook işleme hatası:', error.message);
        return res.status(500).json({ success: false, message: 'Sunucu içi hata oluştu.' });
    }
});
