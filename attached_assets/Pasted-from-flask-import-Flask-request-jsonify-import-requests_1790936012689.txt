from flask import Flask, request, jsonify
import requests

app = Flask(__name__)

# Telegram Bot Token'ınız (Replit Environment Variables üzerinden alınmalıdır)
TOKEN = "BOT_TOKEN_BURaya_GELECEK"
TELEGRAM_API_URL = f"https://api.telegram.org/bot{TOKEN}"

@app.route('/webhook', methods=['POST'])
def webhook():
    update = request.get_json()
    
    # 1. Kullanıcı faturayı onaylamadan önceki son kontrol (Pre-checkout)
    if "pre_checkout_query" in update:
        query_id = update["pre_checkout_query"]["id"]
        # Burada gerekirse stok/hizmet kontrolü yapabilirsiniz
        verify_url = f"{TELEGRAM_API_URL}/answerPreCheckoutQuery"
        requests.post(verify_url, json={"pre_checkout_query_id": query_id, "ok": True})
        return jsonify({"status": "ok"})

    # 2. Ödeme başarıyla gerçekleştiğinde tetiklenen kısım
    if "message" in update and "successful_payment" in update["message"]:
        chat_id = update["message"]["chat"]["id"]
        payment_info = update["message"]["successful_payment"]
        
        # Kullanıcıya hizmetin verildiğine dair onay mesajı gönderilir
        msg = "Ödemeniz başarıyla alındı! MMT Global VIP lojistik kanalına ve verilerine erişiminiz aktif edilmiştir."
        requests.post(f"{TELEGRAM_API_URL}/sendMessage", json={"chat_id": chat_id, "text": msg})
        
        # Burada MMT Connect veritabanına işlem kaydedilebilir ve yetkilendirme tetiklenebilir
        
    return jsonify({"status": "ok"})

# Kullanıcıya Yıldız (Stars) Faturası Gönderme Fonksiyonu
def send_star_invoice(chat_id, title, description, amount_in_stars):
    invoice_url = f"{TELEGRAM_API_URL}/sendInvoice"
    payload = {
        "chat_id": chat_id,
        "title": title,
        "description": description,
        "payload": "mmt_global_service_payload",
        "provider_token": "", # Telegram Stars için provider_token boş bırakılır
        "currency": "XTR",     # XTR, Telegram Stars para birimi kodudur
        "prices": [{"label": title, "amount": amount_in_stars}] # 1 Star = miktar (örn: 100 = 100 Star)
    }
    response = requests.post(invoice_url, json=payload)
    return response.json()

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
