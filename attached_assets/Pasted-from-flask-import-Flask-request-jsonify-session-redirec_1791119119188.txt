@app.route('/')
def index():
    user_key = session.get('user')
    user_data = USERS.get(user_key) if user_key else None
    return render_template_string(HTML_TEMPLATE, user_data=user_data, logs=AUDIT_LOGS)

@app.route('/login', methods=['POST'])
def login():
    username = request.form.get('username')
    password = request.form.get('password')
    if username in USERS and USERS[username]['pass'] == password:
        session['user'] = username
        AUDIT_LOGS.insert(0, {"time": datetime.now().strftime("%Y-%m-%d %H:%M:%S"), "user": USERS[username]['name'], "action": "Sisteme giriş yapıldı."})
    return redirect(url_for('index'))

@app.route('/logout', methods=['POST'])
def logout():
    user_key = session.get('user')
    if user_key:
        AUDIT_LOGS.insert(0, {"time": datetime.now().strftime("%Y-%m-%d %H:%M:%S"), "user": USERS[user_key]['name'], "action": "Oturum kapatıldı."})
    session.pop('user', None)
    return redirect(url_for('index'))

@app.route('/master-action', methods=['POST'])
def master_action():
    user_key = session.get('user')
    if user_key != 'mehmet':
        abort(403, description="Yetkisiz işlem! Bu onay yalnızca İstanbul Master yetkisiyle verilebilir.")

    })
