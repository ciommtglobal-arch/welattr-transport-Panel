import React from 'react';
import { useUser, SignInButton } from '@clerk/clerk-react';

export function StaffSignInForm() {
  const { user, isLoaded, isSignedIn } = useUser();

  if (!isLoaded) {
    return <div className="text-slate-400 p-4 text-center">Yükleniyor...</div>;
  }

  if (!isSignedIn) {
    return (
      <div className="p-6 bg-slate-900 text-white rounded-xl shadow-xl max-w-md mx-auto text-center">
        <h3 className="text-xl font-bold text-orange-500 mb-2">MMT Connect Giriş</h3>
        <p className="text-sm text-slate-300 mb-4">Devam etmek için yetkili hesabınızla oturum açın.</p>
        <SignInButton mode="modal">
          <button className="w-full bg-orange-500 hover:bg-orange-600 py-3 px-4 rounded-lg font-semibold text-white transition cursor-pointer">
            Sisteme Giriş Yap
          </button>
        </SignInButton>
      </div>
    );
  }

  const handleRouteRedirect = (target: 'company' | 'staff') => {
    if (target === 'company') {
      window.location.href = '/b2b-agi/panel';
    } else {
      window.location.href = '/staff/yonlendir';
    }
  };

  return (
    <div className="p-6 bg-slate-900 text-white rounded-xl shadow-xl max-w-md mx-auto">
      <h3 className="text-xl font-bold text-orange-500 mb-2">MMT Connect Harekat Masası</h3>
      <p className="text-sm text-slate-300 mb-4">
        Oturum Açık: <b>{user.fullName || user.primaryEmailAddress?.emailAddress}</b>
      </p>
      <div className="space-y-3">
        <button 
          onClick={() => handleRouteRedirect('company')}
          className="w-full bg-emerald-600 hover:bg-emerald-700 py-3 px-4 rounded-lg font-semibold text-white transition cursor-pointer"
        >
          Şirket Paneline Git (/b2b-agi/panel)
        </button>
        <button 
          onClick={() => handleRouteRedirect('staff')}
          className="w-full bg-blue-600 hover:bg-blue-700 py-3 px-4 rounded-lg font-semibold text-white transition cursor-pointer"
        >
          Personel Paneline Git (/staff/yonlendir)
        </button>
      </div>
    </div>
  );
}
