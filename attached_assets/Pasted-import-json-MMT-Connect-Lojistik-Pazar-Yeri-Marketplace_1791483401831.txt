import json

# MMT Connect - Lojistik Pazar Yeri (Marketplace) Komisyon Kuralları
MARKETPLACE_STANDARDS = {
    "Irak Hattı": {
        "min_komisyon_orani": 5.0, # Sizin alacağınız minimum aracılık komisyonu (%)
        "max_risk_skoru": 5
    },
    "Dubai / BAE Hattı": {
        "min_komisyon_orani": 7.0,
        "max_risk_skoru": 7
    },
    "Avrupa Koridoru": {
        "min_komisyon_orani": 6.0,
        "max_risk_skoru": 4
    },
    "Ukrayna & Doğu Avrupa": {
        "min_komisyon_orani": 8.0,
        "max_risk_skoru": 8
    }
}

def pazar_yeri_aracilik_hesapla(koridor_adi, tirsahibi_maliyeti, musterisatis_fiyati, risk_skoru):
    """
    Yük sahibi ile tır sahibini buluşturduğunuz bu modelde;
    Sizin aradaki aracılık kazancınızı (komisyonunuzu) ve kârınızı hesaplar.
    """
    if koridor_adi not in MARKETPLACE_STANDARDS:
        return {"durum": "HATA", "mesaj": "Geçersiz koridor tanımı."}
    
    standart = MARKETPLACE_STANDARDS[koridor_adi]
    
    # Aradaki brüt kazancınız (Müşteri Satış Fiyatı - Tır Sahibine Ödenen Maliyet)
    aracilik_kazanci = musterisatis_fiyati - tirsahibi_maliyeti
    
    # Sizin komisyon oranınız (Satış üzerindeki payınız)
    if musterisatis_fiyati > 0:
        komisyon_orani = (aracilik_kazanci / musterisatis_fiyati) * 100
    else:
        komisyon_orani = 0.0

    komisyon_uygun = komisyon_orani >= standart["min_komisyon_orani"]
    risk_uygun = risk_skoru <= standart["max_risk_skoru"]
    
    if komisyon_uygun and risk_uygun:
        return {
            "durum": "PAZAR YERİ EŞLEŞMESİ UYGUN — ONAY BEKLENİYOR",
            "aracilik_kazanci_usd": round(aracilik_kazanci, 2),
            "hesaplanan_komisyon_yuzde": round(komisyon_orani, 2),
            "mesaj": f"Alıcı ve satıcı buluştu. Sizin aracılık kazancınız ${round(aracilik_kazanci, 2)} (%{round(komisyon_orani, 2)} komisyon). Ön onay uygun."
        }
    else:
        return {
            "durum": "BLOKE / KOMİSYON DÜŞÜK",
            "aracilik_kazanci_usd": round(aracilik_kazanci, 2),
            "hesaplanan_komisyon_yuzde": round(komisyon_orani, 2),
            "mesaj": f"Komisyon oranı hedef eşiğin altında (%{round(komisyon_orani, 2)}, Min Hedef: %{standart['min_komisyon_orani']})."
        }

if __name__ == "__main__":
    # Örnek: Tır sahibine 2200 USD veriliyor, müşteriye 2800 USD satılıyor. Aradaki fark sizin komisyonunuz.
    sonuc = pazar_yeri_aracilik_hesapla("Irak Hattı", 2200, 2800, 4)
    print(json.dumps(sonuc, ensure_ascii=False, indent=2))
