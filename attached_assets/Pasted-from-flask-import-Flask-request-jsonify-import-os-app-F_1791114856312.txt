from flask import Flask, request, jsonify
import os

app = Flask(__name__)

# MMT Connect - Güvenli Ödeme ve Yetki Yönetim Paneli
# Abdurrahman ağabey teknik/uluslararası operasyonu yönetir, son onay şirket sahibindedir.

@app.route('/api/payment/secure-callback', methods=['POST'])
def secure_payment_callback():
    data = request.json
    
    # 1. Temel Doğrulama ve Güvenli Veri Alımı
    merchant_oid = data.get('merchant_oid')
    status = data.get('status')
    total_amount = data.get('total_amount')
    signature = data.get('signature') # Çelik Kubbe güvenlik imzası
    
    # Güvenlik Kontrolü: İmza veya yetki doğrulanmazsa işlem reddedilir
    if not signature or signature != "CELIK_KUBBE_SECURE_KEY":
        return jsonify({"status": "error", "message": "Güvenlik imzası geçersiz. Çelik Kubbe engelledi."}), 403

    if status == 'success':
        # 2. Para Havuza Düştü - Teknik Operasyon ve Abdurrahman Ağabeyin Paneline Aktarım
        print(f"[HAVUZ] Ödeme Başarılı: Sipariş ID {merchant_oid}, Tutar: {total_amount}")
        print("[OPERASYON] Abdurrahman ağabeyin teknik lojistik kuyruğuna aktarıldı.")
        
        # 3. Son Onay Bekletme Aşaması (Şirket Sahibinin Onayı Olmadan Sevkiyat Yapılmaz)
        return jsonify({
            "status": "pending_owner_approval",
            "message": "Ödeme havuza alındı. Teknik hazırlıklar Abdurrahman ağabey tarafından yürütülüyor. Şirket sahibi onayı bekleniyor."
        }), 200
    else:
        return jsonify({"status": "failed", "message": "Ödeme onaylanmadı."}), 400

@app.route('/api/owner/approve', methods=['POST'])
def owner_approval():
    # Şirket sahibinin son onay mekanizması
    data = request.json
    approval_key = data.get('owner_approval_key')
    merchant_oid = data.get('merchant_oid')
    
    if approval_key == "MEHMET_OGLU_MASTER_KEY":
        print(f"[ONAY] Şirket sahibi tarafından onaylandı. Sipariş ID: {merchant_oid} sevkıyata hazırlanıyor.")
        return jsonify({"status": "success", "message": "Son onay verildi. Sevkiyat süreci başlatıldı."}), 200
    else:
        return jsonify({"status": "error", "message": "Yetkisiz onay denemesi."}), 403

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
