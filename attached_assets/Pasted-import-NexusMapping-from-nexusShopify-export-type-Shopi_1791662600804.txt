import { NexusMapping } from "./nexusShopify";

export type ShopifyRawRequest = {
  rawBody?: Buffer | string;
};

export function parseLogisticsShopifyOrder(
  body: any,
  mapping: NexusMapping,
  topic: string,
) {
  if (!body || (typeof body.id !== "number" && typeof body.id !== "string")) {
    throw new Error("Invalid order payload: missing or invalid order id.");
  }

  const orderIdStr = String(body.id).trim();
  if (!/^[1-9]\d*$/.test(orderIdStr)) {
    throw new Error("Invalid order payload: order id cannot be zero or negative.");
  }
  const orderId = orderIdStr;
  const orderName = body.name && typeof body.name === "string" ? body.name.trim() : `#${orderId}`;

  if (!body.currency || typeof body.currency !== "string" || !/^[A-Z]{3}$/.test(body.currency.trim())) {
    throw new Error("Invalid order payload: missing or invalid ISO currency format.");
  }
  const currency = body.currency.trim();

  if (!body.financial_status || typeof body.financial_status !== "string") {
    throw new Error("Invalid order payload: missing or invalid financial_status.");
  }
  const financialStatus = body.financial_status;

  const cancelled = body.cancelled_at || topic === "orders/cancelled" ? 1 : 0;

  if (!body.updated_at || typeof body.updated_at !== "string") {
    throw new Error("Invalid order payload: missing or invalid updated_at timestamp.");
  }
  const sourceUpdatedAt = new Date(body.updated_at);
  if (isNaN(sourceUpdatedAt.getTime())) {
    throw new Error("Invalid order payload: invalid updated_at timestamp format.");
  }

  // Çelişen koridor kontrolü: note_attributes ve body.corridor kıyaslaması
  const noteAttributes = Array.isArray(body.note_attributes) ? body.note_attributes : [];
  const corridorAttrs = noteAttributes.filter((attr: any) => attr && (attr.name === "corridor" || attr.name === "koridor"));
  
  if (corridorAttrs.length > 1) {
    const firstVal = String(corridorAttrs[0].value).trim();
    if (corridorAttrs.some((attr: any) => String(attr.value).trim() !== firstVal)) {
      throw new Error("Invalid order payload: conflicting corridor definitions found in note_attributes.");
    }
  }

  const attrCorridor = corridorAttrs.length > 0 ? String(corridorAttrs[0].value).trim() : "";
  const bodyCorridor = body.corridor !== undefined && body.corridor !== null ? String(body.corridor).trim() : "";

  if (attrCorridor && bodyCorridor && attrCorridor !== bodyCorridor) {
    throw new Error("Invalid order payload: conflict between note_attributes corridor and body.corridor.");
  }

  const rawCorridor = attrCorridor || bodyCorridor;
  const allowedCorridors = ["Samsun-Irak-Proje", "Samsun-Gurcistan"];
  
  if (!rawCorridor || !allowedCorridors.includes(rawCorridor)) {
    throw new Error(`Invalid order payload: corridor '${rawCorridor}' is not permitted or missing.`);
  }
  const corridor = rawCorridor;

  const lineItems = body.line_items;
  if (!Array.isArray(lineItems)) {
    throw new Error("Invalid order payload: line_items must be an array.");
  }

  if (lineItems.length === 0 && (topic === "orders/create" || topic === "orders/paid")) {
    throw new Error(`Invalid order payload: new or paid orders (${topic}) cannot have empty line items.`);
  }

  const rows = [];
  const seenLineIds = new Set<string>();

  for (const item of lineItems) {
    const lineIdStr = item.id !== undefined && item.id !== null ? String(item.id).trim() : "";
    if (!lineIdStr || lineIdStr === "undefined" || !/^[1-9]\d*$/.test(lineIdStr) || seenLineIds.has(lineIdStr)) {
      throw new Error("Invalid order payload: missing, undefined, zero, negative, or duplicate line item id.");
    }
    seenLineIds.add(lineIdStr);
    const lineId = lineIdStr;

    if (item.variant_id === undefined || item.variant_id === null || String(item.variant_id).trim() === "" || String(item.variant_id) === "0") {
      throw new Error(`Invalid order payload: line item ${lineId} is missing a valid variant_id.`);
    }
    const variantId = String(item.variant_id);

    if (!item.title || typeof item.title !== "string") {
      throw new Error(`Invalid order payload: missing product title for line item ${lineId}.`);
    }
    const productTitle = item.title;

    const rawQuantity = item.current_quantity !== undefined ? item.current_quantity : item.quantity;
    const qtyStr = String(rawQuantity ?? "").trim();
    if (!/^\d+$/.test(qtyStr)) {
      throw new Error(`Invalid order payload: non-integer or malformed quantity for line item ${lineId}.`);
    }
    const quantity = Number(qtyStr);
    if (quantity < 0 || quantity > 1000000) {
      throw new Error(`Invalid order payload: quantity out of bounds for line item ${lineId}.`);
    }

    const rawPrice = item.price;
    if (rawPrice === undefined || rawPrice === null || String(rawPrice).trim() === "") {
      throw new Error(`Invalid order payload: missing price for line item ${lineId}.`);
    }
    const priceStr = String(rawPrice).trim();
    if (!/^\d+(\.\d+)?$/.test(priceStr)) {
      throw new Error(`Invalid order payload: malformed price format for line item ${lineId}.`);
    }
    const numericPrice = Number(priceStr);
    if (!Number.isFinite(numericPrice) || numericPrice < 0 || numericPrice > 100000000) {
      throw new Error(`Invalid order payload: invalid or out-of-bounds unit price for line item ${lineId}.`);
    }
    const unitPrice = numericPrice.toFixed(2);

    let kind = "reservation";
    if (mapping.reservations.includes(variantId)) {
      kind = "reservation";
    } else if (mapping.financing.includes(variantId)) {
      kind = "financing_interest";
    }

    rows.push({
      shopDomain: mapping.shopDomain,
      orderId,
      lineId,
      orderName,
      variantId,
      kind,
      productTitle,
      quantity,
      unitPrice,
      currency,
      financialStatus,
      cancelled,
      warning: `Corridor: ${corridor}`,
      sourceUpdatedAt,
      status: "PENDING_FOUNDER_APPROVAL" as const,
    });
  }

  return {
    shopDomain: mapping.shopDomain,
    orderId,
    orderName,
    currency,
    financialStatus,
    cancelled,
    sourceUpdatedAt,
    warning: `Processed successfully for corridor: ${corridor}`,
    rows,
  };
}
