/**
 * MMT Connect & Shopify Güvenli Entegrasyon Modülü
 * Çelik Kubbe & WelatTR İmza Protokolü
 */

const crypto = require('crypto');
const express = require('express');
const router = express.Router();

// Shopify Webhook HMAC İmza Doğrulama Fonksiyonu
function verifyShopifyWebhook(req, res, next) {
    const hmacHeader = req.get('X-Shopify-Hmac-Sha256');
    const secret = process.env.SHOPIFY_WEBHOOK_SECRET; // Çelik Kubbe Güvenlik Anahtarı
    
    if (!hmacHeader || !secret) {
        return res.status(401).json({ status: 'error', message: 'Çelik Kubbe: Güvenlik anahtarı veya imza eksik.' });
    }

    const calculatedHmac = crypto
        .createHmac('sha256', secret)
        .update(JSON.stringify(req.body))
        .digest('base64');

    if (calculatedHmac !== hmacHeader) {
        return res.status(403).json({ status: 'error', message: 'Çelik Kubbe: Yetkisiz imza tespiti, erişim engellendi.' });
    }
    next();
}

// Güvenli Webhook Endpoint (Kurucu Onayı Bekliyor Modu)
router.post('/api/mmt/secure-sync', verifyShopifyWebhook, async (req, res) => {
    try {
        const { product_id, title, inventory_quantity, corridor } = req.body;

        // Samsun - Irak Proje ve Boru Hattı Filtrelemesi
        if (corridor !== 'Samsun-Irak-Proje') {
            return res.status(400).json({ 
                status: 'error', 
                message: 'Geçersiz koridor tanımı.' 
            });
        }

        // Simüle Edilmiş Veritabanı Kaydı (Kurucu Onayı Şart)
        const pendingRecord = {
            timestamp: new Date().toISOString(),
            item: title,
            stock: inventory_quantity,
            status: 'Kurucu Onayı Bekliyor',
            authorizedBy: 'WelatTR (Beklemede)'
        };

        console.log(`[MMT Nexus] Yeni talep havuzda: ${title} - Onay bekleniyor.`);

        return res.status(200).json({
            status: 'success',
            message: 'Talep başarıyla alındı. Çelik Kubbe filtresinden geçti ve Kurucu Onay Bekliyor kuyruğuna eklendi.',
            data: pendingRecord
        });

    } catch (error) {
        console.error('Entegrasyon Hatası:', error.message);
        return res.status(500).json({ status: 'error', message: error.message });
    }
});

module.exports = router;
