import json

# Çelik Kubbe ve Uluslararası Koridor Veritabanı
CORRIDOR_STANDARDS = {
    "Irak Hattı": {
        "hedef_marj": 18.0,
        "max_risk_skoru": 5,
        "piyasa_referans_satis": 2800, # USD (Müşteriye verilecek minimum referans satış fiyatı)
        "para_birimi": "USD",
        "ortalama_sure_gun": 3
    },
    "Dubai / BAE Hattı": {
        "hedef_marj": 22.0,
        "max_risk_skoru": 7,
        "piyasa_referans_satis": 6500,
        "para_birimi": "USD",
        "ortalama_sure_gun": 12
    },
    "Avrupa Koridoru": {
        "hedef_marj": 20.0,
        "max_risk_skoru": 4,
        "piyasa_referans_satis": 5200,
        "para_birimi": "USD",
        "ortalama_sure_gun": 8
    },
    "Ukrayna & Doğu Avrupa": {
        "hedef_marj": 25.0,
        "max_risk_skoru": 8,
        "piyasa_referans_satis": 4800,
        "para_birimi": "USD",
        "ortalama_sure_gun": 7
    }
}

def celik_kubbe_maliyet_satis_analizi(koridor_adi, tasima_maliyeti, satis_fiyati, risk_skoru):
    """
    Taşıma maliyeti (tedarikçi gideri) ile satış fiyatını (müşteri geliri) 
    ayrı ayrı ele alarak gerçek kâr marjını ve Çelik Kubbe ön kontrolünü yapar.
    """
    if koridor_adi not in CORRIDOR_STANDARDS:
        return {"durum": "HATA", "mesaj": "Geçersiz koridor tanımı."}
    
    standart = CORRIDOR_STANDARDS[koridor_adi]
    
    # Gerçek kâr marjı hesabi: ((Satış - Maliyet) / Satış) * 100
    if satis_fiyati > 0:
        hesaplanan_marj = ((satis_fiyati - tasima_maliyeti) / satis_fiyati) * 100
    else:
        hesaplanan_marj = 0.0

    # Kriter Kontrolleri
    marj_uygun = hesaplanan_marj >= standart["hedef_marj"]
    risk_uygun = risk_skoru <= standart["max_risk_skoru"]
    satis_uygun = satis_fiyati >= standart["piyasa_referans_satis"]
    
    if marj_uygun and risk_uygun and satis_uygun:
        return {
            "durum": "ÖN KONTROL UYGUN — KURUCU ONAYI BEKLENİYOR",
            "hesaplanan_marj_yuzde": round(hesaplanan_marj, 2),
            "mesaj": f"Gerçek kâr marjı %{round(hesaplanan_marj, 2)} (Hedef: %{standart['hedef_marj']}). Ön kontrol uygun; son onay için CIO / WelatTR bekleniyor."
        }
    else:
        sebepler = []
        if not marj_uygun:
            sebepler.append(f"Gerçek kâr marjı yetersiz (%{round(hesaplanan_marj, 2)}, Hedef: %{standart['hedef_marj']})")
        if not risk_uygun:
            sebepler.append(f"Risk skoru yüksek ({risk_skoru}, Max: {standart['max_risk_skoru']})")
        if not satis_uygun:
            sebepler.append(f"Satış fiyatı piyasa referansının altında (${satis_fiyati}, Referans: ${standart['piyasa_referans_satis']})")
            
        return {
            "durum": "BLOKE / CIO ONAYINA DÜŞTÜ",
            "hesaplanan_marj_yuzde": round(hesaplanan_marj, 2),
            "mesaj": "Çelik Kubbe risk uyarısı: " + " | ".join(sebepler)
        }

if __name__ == "__main__":
    # Örnek: Irak hattı için 2000 USD maliyet, 2800 USD satış fiyatı testi
    sonuc = celik_kubbe_maliyet_satis_analizi("Irak Hattı", 2000, 2800, 4)
    print(json.dumps(sonuc, ensure_ascii=False, indent=2))
