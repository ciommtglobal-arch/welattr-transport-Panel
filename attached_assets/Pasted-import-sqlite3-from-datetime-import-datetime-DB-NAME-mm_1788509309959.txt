import sqlite3
from datetime import datetime

DB_NAME = "mmt_alsat_test.db"

def veritabani_hazirla():
    with sqlite3.connect(DB_NAME) as conn:
        cursor = conn.cursor()
        
        # Kullanıcı Cüzdanı (Nakit USD ve Sahip Olduğu Navlun Hissesi)
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS cuzdanlar (
                user_id INTEGER PRIMARY KEY,
                ad_soyad TEXT,
                nakit_usd REAL DEFAULT 0.0,
                navlun_hissesi INTEGER DEFAULT 0
            )
        """)
        
        # Platform Kasa Tablosu (MMT Global Komisyon Havuzu)
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS platform_kasa (
                id INTEGER PRIMARY KEY,
                toplam_komisyon_usd REAL DEFAULT 0.0
            )
        """)
        
        # İşlem Geçmişi Defteri
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS islem_defteri (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER,
                islem_turu TEXT,
                adet INTEGER,
                birim_fiyat REAL,
                brut_tutar REAL,
                kesilen_komisyon REAL,
                net_tutar REAL,
                tarih TEXT
            )
        """)
        
        # Platform kasasını başlat (1 nolu kayıt yoksa ekle)
        cursor.execute("INSERT OR IGNORE INTO platform_kasa (id, toplam_komisyon_usd) VALUES (1, 0.0)")
        conn.commit()

class MMTTicaretMotoru:
    def __init__(self, komisyon_orani: float = 0.02):
        """
        komisyon_orani: İşlem başına alınan komisyon yüzdesi. 
        Örn: 0.02 = %2 MMT aracılık komisyonu.
        """
        self.komisyon_orani = komisyon_orani
        veritabani_hazirla()

    def kullanici_olustur_veya_getir(self, user_id: int, ad_soyad: str, baslangic_nakit: float = 10000.0):
        with sqlite3.connect(DB_NAME) as conn:
            cursor = conn.cursor()
            cursor.execute("SELECT * FROM cuzdanlar WHERE user_id = ?", (user_id,))
            kullanici = cursor.fetchone()
            if not kullanici:
                cursor.execute(
                    "INSERT INTO cuzdanlar VALUES (?, ?, ?, 0)",
                    (user_id, ad_soyad, baslangic_nakit)
                )
                conn.commit()
                return {"user_id": user_id, "ad_soyad": ad_soyad, "nakit_usd": baslangic_nakit, "navlun_hissesi": 0}
            return {"user_id": kullanici[0], "ad_soyad": kullanici[1], "nakit_usd": kullanici[2], "navlun_hissesi": kullanici[3]}

    def navlun_payi_al(self, user_id: int, adet: int, hisse_fiyati: float):
        """
        Navlun hissesi alımı simülasyonu.
        Alıcı brüt tutar + komisyon öder (veya brüt içerisinden platforma kesilir).
        """
        brut_maliyet = adet * hisse_fiyati
        komisyon = brut_maliyet * self.komisyon_orani
        toplam_tutar = brut_maliyet + komisyon

        with sqlite3.connect(DB_NAME) as conn:
            cursor = conn.cursor()
            cursor.execute("SELECT nakit_usd, navlun_hissesi FROM cuzdanlar WHERE user_id = ?", (user_id,))
            cuzdan = cursor.fetchone()
            
            if not cuzdan:
                return False, "Kullanıcı bulunamadı."
            if cuzdan[0] < toplam_tutar:
                return False, f"Yetersiz bakiye! Gereken: ${toplam_tutar:,.2f}, Mevcut: ${cuzdan[0]:,.2f}"

            # Kullanıcıdan nakit düş, hisse ekle
            cursor.execute("""
                UPDATE cuzdanlar 
                SET nakit_usd = nakit_usd - ?, navlun_hissesi = navlun_hissesi + ? 
                WHERE user_id = ?
            """, (toplam_tutar, adet, user_id))

            # Platform kasasına komisyon ekle
            cursor.execute("UPDATE platform_kasa SET toplam_komisyon_usd = toplam_komisyon_usd + ? WHERE id = 1", (komisyon,))

            # İşlem defterine yaz
            tarih = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            cursor.execute("""
                INSERT INTO islem_defteri (user_id, islem_turu, adet, birim_fiyat, brut_tutar, kesilen_komisyon, net_tutar, tarih)
                VALUES (?, 'ALIS', ?, ?, ?, ?, ?, ?)
            """, (user_id, adet, hisse_fiyati, brut_maliyet, komisyon, toplam_tutar, tarih))

            conn.commit()
            return True, f"{adet} adet hisse ${toplam_tutar:,.2f} maliyetle alındı (${komisyon:,.2f} MMT komisyonu)."

    def navlun_payi_sat(self, user_id: int, adet: int, hisse_fiyati: float):
        """
        Navlun hissesi satışı simülasyonu.
        Satış tutarından komisyon kesilir, kalan net nakit kullanıcıya ödenir.
        """
        with sqlite3.connect(DB_NAME) as conn:
            cursor = conn.cursor()
            cursor.execute("SELECT nakit_usd, navlun_hissesi FROM cuzdanlar WHERE user_id = ?", (user_id,))
            cuzdan = cursor.fetchone()

            if not cuzdan:
                return False, "Kullanıcı bulunamadı."
            if cuzdan[1] < adet:
                return False, f"Yetersiz hisse! Mevcut: {cuzdan[1]} adet, Satılmak istenen: {adet} adet."

            brut_gelir = adet * hisse_fiyati
            komisyon = brut_gelir * self.komisyon_orani
            net_odenecek = brut_gelir - komisyon

            # Kullanıcıdan hisse düş, net nakit ekle
            cursor.execute("""
                UPDATE cuzdanlar 
                SET nakit_usd = nakit_usd + ?, navlun_hissesi = navlun_hissesi - ? 
                WHERE user_id = ?
            """, (net_odenecek, adet, user_id))

            # Platform kasasına komisyon ekle
            cursor.execute("UPDATE platform_kasa SET toplam_komisyon_usd = toplam_komisyon_usd + ? WHERE id = 1", (komisyon,))

            # İşlem defterine yaz
            tarih = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            cursor.execute("""
                INSERT INTO islem_defteri (user_id, islem_turu, adet, birim_fiyat, brut_tutar, kesilen_komisyon, net_tutar, tarih)
                VALUES (?, 'SATIS', ?, ?, ?, ?, ?, ?)
            """, (user_id, adet, hisse_fiyati, brut_gelir, komisyon, net_odenecek, tarih))

            conn.commit()
            return True, f"{adet} adet hisse satıldı. Net yatan: ${net_odenecek:,.2f} (${komisyon:,.2f} MMT komisyonu kesildi)."

    def platform_raporu(self):
        with sqlite3.connect(DB_NAME) as conn:
            cursor = conn.cursor()
            komisyon = cursor.execute("SELECT toplam_komisyon_usd FROM platform_kasa WHERE id = 1").fetchone()[0]
            islem_sayisi = cursor.execute("SELECT COUNT(*) FROM islem_defteri").fetchone()[0]
            return {"toplam_komisyon": komisyon, "toplam_islem": islem_sayisi}
