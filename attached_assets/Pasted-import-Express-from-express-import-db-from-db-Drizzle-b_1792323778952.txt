import { Express } from "express";
import { db } from "./db"; // Drizzle bağlantınız
import { cargoPool, emptyTrucks } from "@shared/schema";
import { eq, and } from "drizzle-orm";

export function registerLogisticsRoutes(app: Express) {
  
  // 1. Güney Bölgesi Yük Kaydı Ekle (Mersin, İskenderun, Antep, Adana)
  app.post("/api/cargo/add", async (req, res) => {
    try {
      const { region, loadDetails, weightTon, destination } = req.body;
      const [newCargo] = await db.insert(cargoPool).values({
        region,
        loadDetails,
        weightTon,
        destination,
      }).returning();
      
      res.json({ success: true, message: "Yük başarıyla havuza eklendi.", data: newCargo });
    } catch (error) {
      res.status(500).json({ success: false, error: error.message });
    }
  });

  // 2. Mardin/Urfa Boş Araç Kaydı Ekle
  app.post("/api/trucks/empty", async (req, res) => {
    try {
      const { originRegion, targetDestination, plateOrContact } = req.body;
      const [newTruck] = await db.insert(emptyTrucks).values({
        originRegion,
        targetDestination,
        plateOrContact,
      }).returning();

      res.json({ success: true, message: "Boş araç arzı kaydedildi.", data: newTruck });
    } catch (error) {
      res.status(500).json({ success: false, error: error.message });
    }
  });

  // 3. Havuz Verilerini Listele (Önce veri toplama modu)
  app.get("/api/logistics/pool-status", async (req, res) => {
    try {
      const activeCargoes = await db.select().from(cargoPool).where(eq(cargoPool.status, "active"));
      const availableTrucksList = await db.select().from(emptyTrucks).where(eq(emptyTrucks.status, "available"));

      res.json({
        success: true,
        cargoCount: activeCargoes.length,
        truckCount: availableTrucksList.length,
        cargoes: activeCargoes,
        trucks: availableTrucksList
      });
    } catch (error) {
      res.status(500).json({ success: false, error: error.message });
    }
  });

}
