import { NexusMapping } from "./nexusShopify";
import { Request, Response, NextFunction } from "express";

export type ShopifyRawRequest = Request & {
  rawBody?: Buffer | string;
};

// Sertan'ın uyardığı ham gövde middleware ve raw parser sarmalayıcısı
export function createShopifyLogisticsRawParser(mapping: NexusMapping) {
  return (req: ShopifyRawRequest, res: Response, next: NextFunction) => {
    try {
      const topic = req.get("X-Shopify-Topic") ?? "";
      const raw = req.rawBody;

      if (!raw) {
        return res.status(400).json({ error: "Missing raw request body for signature verification." });
      }

      const body = JSON.parse(typeof raw === "string" ? raw : raw.toString("utf8"));
      
      // Siparişi ayrıştır ve request.body içine ata
      req.body = parseLogisticsShopifyOrder(body, mapping, topic);
      next();
    } catch (error: any) {
      return res.status(400).json({ error: `Invalid logistics payload: ${error.message}` });
    }
  };
}
