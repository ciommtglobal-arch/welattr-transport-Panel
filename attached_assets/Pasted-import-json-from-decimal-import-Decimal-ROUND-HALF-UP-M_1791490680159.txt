import json
from decimal import Decimal, ROUND_HALF_UP

# --- MMT GLOBAL - ÇELİK KUBBE OTONOM GÜVENLİK VE HESAP MOTORU ---
SISTEM_KAYIT = {
    "proje": "MMT Connect & Çelik Kubbe",
    "kurucu": "Mehmet Alioğlu (CIO / WelatTR)",
    "guvenlik_modu": "Kurucu Onaylı Taslak Denetimi"
}

MARKETPLACE_TARIFF = {
    "Irak": {"toplam_komisyon_orani": Decimal("6.0"), "max_risk": 5},
    "Dubai / BAE": {"toplam_komisyon_orani": Decimal("8.0"), "max_risk": 7},
    "Avrupa": {"toplam_komisyon_orani": Decimal("7.0"), "max_risk": 4},
    "Ukrayna / Doğu Avrupa": {"toplam_komisyon_orani": Decimal("10.0"), "max_risk": 8}
}

def mmt_guvenli_hesapla_ve_denetle(koridor, musteri_satis_fiyati, tirsahibi_maliyeti, risk_skoru):
    """
    1. Kesirli risk ve geçersiz biçim denetimlerini yapar.
    2. Çelik Kubbe risk eşiğini kontrol eder.
    3. Decimal ile kuruş hatasız, net hakediş ve ticari fark dengesini kurar.
    4. Negatif net hakedişleri bloke eder ve kurucu onayına sunar.
    """
    try:
        satis = Decimal(str(musteri_satis_fiyati))
        maliyet = Decimal(str(tirsahibi_maliyeti))
        risk = int(risk_skoru)
    except (ValueError, TypeError):
        return {"durum": "HATA", "mesaj": "Geçersiz veri biçimi veya sayısal değer."}
    
    if satis <= Decimal("0.00") or maliyet <= Decimal("0.00") or risk < 0:
        return {"durum": "REDDEDİLDİ", "mesaj": "Sıfır, negatif fiyat veya geçersiz risk değeri."}
    
    if koridor not in MARKETPLACE_TARIFF:
        return {"durum": "HATA", "mesaj": "Tanımsız lojistik koridoru."}
    
    kural = MARKETPLACE_TARIFF[koridor]
    
    # Çelik Kubbe Risk Süzgeci
    if risk > kural["max_risk"]:
        return {
            "durum": "BLOKE / ÇELİK KUBBE RİSK AŞIMI",
            "mesaj": f"Risk skoru ({risk}) izin verilen maksimum sınırın ({kural['max_risk']}) üzerinde."
        }
    
    # Komisyon ve Marj Hesapları (Decimal Kuruş Hassasiyeti)
    toplam_komisyon = (satis * (kural["toplam_komisyon_orani"] / Decimal("100.0"))).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
    tasiyici_komisyon_payi = (toplam_komisyon / Decimal("2.0")).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
    
    # Net Hakediş ve Negatif Kontrolü
    tir_sahibi_net_hakedis = maliyet - tasiyici_komisyon_payi
    if tir_sahibi_net_hakedis < Decimal("0.00"):
        return {"durum": "BLOKE", "mesaj": "Hata: Taşıyıcı net hakedişi negatif olamaz."}
    
    brut_ticari_satis_farki = satis - maliyet
    
    return {
        "durum": "TASLAK HESAP - KURUCU ONAYI BEKLENİYOR",
        "sistem_bilgisi": SISTEM_KAYIT,
        "detaylar": {
            "koridor": koridor,
            "musteri_satis_fiyati_usd": float(satis),
            "tasiyici_net_hakedis_usd": float(tir_sahibi_net_hakedis),
            "mmt_platform_komisyonu_usd": float(toplam_komisyon),
            "mmt_brut_ticari_satis_farki_usd": float(brut_ticari_satis_farki),
            "matematiksel_denge_kontrolu": float(satis) == float(tir_sahibi_net_hakedis + tasiyici_komisyon_payi + brut_ticari_satis_farki)
        }
    }

if __name__ == "__main__":
    # Test Simülasyonu (Örn: Irak hattı, 3000 USD satış, 2700 USD maliyet, 3 risk)
    sonuc = mmt_guvenli_hesapla_ve_denetle("Irak", 3000.0, 2700.0, 3)
    print(json.dumps(sonuc, ensure_ascii=False, indent=2))
