from telegram import Update
from telegram.ext import ContextTypes

# Sistemde tanımlı ve doğrulanmış kurucuya ait Telegram User ID
KURUCU_TELEGRAM_ID = 123456789  # Örnek ID, gerçek ID ile güncellenecektir

async def buton_tiklama_isleyicisi(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    
    # 1. Buton verisinin güvenli parçalanması (split ile ilk ayrım)
    if not query.data or "_" not in query.data:
        await query.edit_message_text(text="❌ Geçersiz işlem verisi.")
        return
        
    islem_turu, islem_id = query.data.split("_", 1)
    
    # 2. Kurucu Yetkisi Doğrulaması (Lider veya asistan son onayı veremez)
    tikalayan_kullanici_id = query.from_user.id
    if tikalayan_kullanici_id != KURUCU_TELEGRAM_ID:
        await query.edit_message_text(text="❌ Bu işlemi onaylamaya yetkiniz yok. Yalnızca Kurucu onay verebilir.")
        return
        
    # 3. Veritabanında işlemin tekrar kontrolü (Re-check)
    # islem = db.islem_getir(islem_id)
    # if not islem or islem['durum'] != 'TASLAK HESAP - KURUCU ONAYI BEKLENİYOR':
    #     await query.edit_message_text(text="❌ Bu işlem artık onay bekliyor durumunda değil veya zaten sonuçlandırılmış.")
    #     return

    if islem_turu == "onayla":
        # 4. Matematiksel denge ve veritabanı kayıt işlemi (Transaction içinde)
        # basarili_mi = db.islemi_onayla(islem_id, onaylayan="Kurucu")
        basarili_mi = True  # Simülasyon
        
        if basarili_mi:
            await query.edit_message_text(
                text=f"✅ İşlem ID `{islem_id}` matematiksel olarak doğrulandı ve **Kurucu Onayı** ile kesinleştirildi."
            )
            # Not: Banka/cüzdan aktarımı bu adımdan tamamen bağımsız ayrı bir tetikleyiciye bağlıdır.
        else:
            await query.edit_message_text(text="❌ Kayıt sırasında bir hata oluştu, işlem gerçekleştirilemedi.")
            
    elif islem_turu == "reddet":
        # db.islemi_reddet(islem_id)
        await query.edit_message_text(text=f"❌ İşlem ID `{islem_id}` **reddedildi**.")
